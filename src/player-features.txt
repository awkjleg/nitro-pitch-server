(function () {
  'use strict';
  const text=(value,max=40)=>typeof value==='string'?value.slice(0,max):'';
  const number=(value,max=1000000)=>Number.isFinite(value)?Math.max(0,Math.min(max,Math.round(value))):0;
  function history(value){
    if(!Array.isArray(value))return[];
    return value.filter(row=>row&&typeof row.id==='string'&&row.id.length>0&&row.id.length<=80&&Number.isFinite(row.at)&&Array.isArray(row.score)&&row.score.length===2).slice(0,20).map(row=>({id:row.id,at:number(row.at,1e15),title:text(row.title,80),result:['Win','Loss','Draw'].includes(row.result)?row.result:'Draw',score:row.score.map(v=>number(v,999)),goals:number(row.goals),assists:number(row.assists),saves:number(row.saves),xp:number(row.xp),team:Array.isArray(row.team)?row.team.slice(0,6).map(v=>text(v)):[],opponents:Array.isArray(row.opponents)?row.opponents.slice(0,6).map(v=>text(v)):[],rank:row.rank&&Number.isFinite(row.rank.before)&&Number.isFinite(row.rank.after)?{before:number(row.rank.before,4000),after:number(row.rank.after,4000)}:null}));
  }
  function addHistory(rows,row){return history([row,...history(rows).filter(item=>item.id!==row.id)]);}
  function identity(player){const uid=player&&(player.ui||player.uid||player.pr?.ui),peer=player&&(player.peer||player.key);return typeof uid==='string'&&/^[A-Za-z0-9_-]{1,128}$/.test(uid)?'uid:'+uid:typeof peer==='string'&&/^[A-Za-z0-9_-]{1,128}$/.test(peer)?'peer:'+peer:'';}
  function blocks(value){return Array.isArray(value)?value.filter(item=>item&&/^(uid|peer):[A-Za-z0-9_-]{1,128}$/.test(item.id)&&typeof item.name==='string').slice(0,100).map(item=>({id:item.id,name:text(item.name),at:number(item.at,1e15)})):[];}
  function rematch(mid,expected,peers){
    const ids=[...new Set(expected)].sort();
    const players=peers.filter(p=>p.presence&&Number.isFinite(p.presence.j)&&[0,1].includes(p.presence.tm)&&!p.presence.sp);
    const current=[...new Set(players.map(p=>p.peer))].sort();
    const complete=ids.length>=2&&JSON.stringify(ids)===JSON.stringify(current);
    const votes=players.filter(p=>ids.includes(p.peer)&&p.presence.npRematch?.mid===mid&&p.presence.npRematch.yes===true&&Array.isArray(p.presence.npRematch.roster)&&JSON.stringify([...p.presence.npRematch.roster].sort())===JSON.stringify(ids)).length;
    return{total:ids.length,votes,complete,ready:complete&&votes===ids.length};
  }
  window.NPPlayerCore=Object.freeze({history,addHistory,identity,blocks,rematch});
})();

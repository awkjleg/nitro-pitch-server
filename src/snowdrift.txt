}

window.NPProgressAPI?.listen(data => {
  const before = snowEv ? snowEv.hd | 0 : null;
  if (data && data.ev) snowEv = data.ev;
  if (snowLive() && before != null && snowEv && (snowEv.got | 0) > 0 && (snowEv.hd | 0) > before && !snowBusy) toast("Christmas Drop earned! Open it in Nitro Pass › Snowdrift");
  if (screen === "pass" && passTab === "snow") setTimeout(() => renderPass(), 30);
});
let snowBusy = false;
async function snowCall(action, data) {
  snowBusy = true;
  try { const r = await window.NPProgressAPI.call(action, { id: crypto.randomUUID(), ...data }); if (r.profile && r.profile.ev) snowEv = r.profile.ev; return r; }
  finally { snowBusy = false; }
}

function snowShowGot(title, g) {
  if (g && g.kind) { const it = listOf(g.kind)[g.i]; snowShow(title, itemThumb(g.kind, g.i), rarCol(it), it.name, rarName(it) + " · " + kindName(g.kind), g); }
  else if (g && g.xp) snowShow(title, h("b", "np-xp", "+" + g.xp.toLocaleString()), "#ffd451", "+" + g.xp.toLocaleString() + " XP", g.had ? "" : "");
  else if (g && g.drops) snowShow(title, snowGiftThumb(), "#ff4d57", g.drops + " Christmas Drop" + (g.drops > 1 ? "s" : ""), g.had ? "You already had " + g.had + ", so here's a drop instead" : "Open them in Nitro Pass › Snowdrift");
}

  const A = window.npAccounts; if (!A || !A.user()) { toast("Log in to open Christmas Drops"); return; }
  try { const r = await snowCall("holiday-drop"); snowShowGot("Christmas Drop", r.result.got); }
  catch (e) { toast((e && e.message) || "Couldn't open the drop"); }
}

const snowToday = () => Math.floor((Date.now() - snowData().at) / 864e5) + 1;
function snowDoorsOpen() { if (!snowLive() || !snowEv) return 0; let n = 0; for (let d = 1; d <= Math.min(25, snowToday()); d++) if (!(snowEv.adv & (1 << (d - 1)))) n++; return n; }

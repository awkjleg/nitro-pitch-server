async function shareReplay(id) {
  const A = window.npAccounts;
  if (!id) { toast("Save the replay first, then share it from Saved Replays"); return; }
  if (!A || !A.user()) { toast("Log in to share replays"); return; }
  let data = null; try { data = localStorage.getItem("np-rp-" + id); } catch {}
  if (!data) { toast("That replay couldn't be found"); return; }
  const it = rpIndex().find(x => x.id === id);
  toast("Making a link…");
  try {
    const token = await A.token(false);
    const r = await fetch("/api/clips", { method: "POST", headers: { "Content-Type": "application/json", Authorization: "Bearer " + token }, body: JSON.stringify({ data, title: it ? it.title : "Replay" }) });
    const j = await r.json().catch(() => ({}));
    if (!r.ok) throw new Error(j.error || "Couldn't share the replay");
    const link = location.origin + "/?clip=" + j.id;
    let copied = false; try { await navigator.clipboard.writeText(link); copied = true; } catch {}
    if (copied) toast("Link copied! Anyone can open it for 30 days: " + link);
    else prompt("Copy this link (it works for 30 days):", link);
  } catch (e) { toast((e && e.message) || "Couldn't share the replay"); }
}

// ---- chat moderation: swearing or spamming in lobby chat mutes you for a while (growing each time)
const AM_KEY = "np-automute";
function amState() { try { const s = JSON.parse(localStorage.getItem(AM_KEY) || "null"); return s && typeof s === "object" ? { until: +s.until || 0, strikes: +s.strikes || 0, at: +s.at || 0 } : { until: 0, strikes: 0, at: 0 }; } catch { return { until: 0, strikes: 0, at: 0 }; } }
function amSave(s) { try { localStorage.setItem(AM_KEY, JSON.stringify(s)); } catch {} }
const amRecent = [];
{
  const form = $("lchatForm"), submit = form.onsubmit;
  form.onsubmit = function (e) {
    const now = Date.now(), s = amState(), raw = $("lchatIn").value;
    if (s.at && now - s.at > 864e5) { s.strikes = 0; s.at = 0; amSave(s); }       // strikes are forgotten after a day
    if (s.until > now && lchatCanSend()) { e.preventDefault(); lchatNote("You're auto-muted for " + muteLeftText(s.until - now) + " for breaking the chat rules.", true); return; }
    const clean = lchatClean(raw), norm = clean.toLowerCase().replace(/[^\p{L}\p{N}]/gu, "");
    LCHAT_SWEARS.lastIndex = 0;
    const swear = LCHAT_SWEARS.test(clean); LCHAT_SWEARS.lastIndex = 0;
    while (amRecent.length && now - amRecent[0].t > 60000) amRecent.shift();
    const repeats = norm.length > 0 && amRecent.filter(x => x.n === norm).length >= 2;
    const shout = clean.replace(/[^A-Z]/g, "").length >= 12 && clean.replace(/[^A-Z]/g, "").length > clean.replace(/[^a-z]/g, "").length * 3;
    if ((swear || repeats || shout) && lchatCanSend() && (lchat.tab || "lobby") !== "clan") {
      e.preventDefault();
      s.strikes++; s.at = now;
      const why = swear ? "Keep the chat friendly: no swearing." : repeats ? "Don't spam the same message." : "Please don't shout in all caps.";
      if (s.strikes >= 2) { const m = Math.min(30, 2 ** (s.strikes - 2) * 2); s.until = now + m * 60000; lchatNote(why + " You're auto-muted for " + m + " minutes.", true); }
      else lchatNote(why + " Next time you'll be muted.", true);
      amSave(s); return;
    }
    if (norm) amRecent.push({ t: now, n: norm });
    return submit.call(this, e);
  };
}

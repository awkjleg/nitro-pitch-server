  // ---------- names: "That name isn't allowed" and the admin's lock on the rename box ----------
  let nameLock = null;
  const lockSvg = '<svg viewBox="0 0 24 24" aria-hidden="true"><rect x="5" y="10.5" width="14" height="10" rx="1.5"/><path d="M8 10.5V7.5a4 4 0 0 1 8 0v3M12 14.5v2.5"/></svg>';
  function renderLock() {
    const nick = $("nick"); if (!nick) return;
    const on = !!nameLock && signedIn();
    nick.readOnly = on; nick.classList.toggle("np-locked", on);
    nick.title = on ? "Your name was set by an admin and is locked" : "";
    let mark = document.getElementById("npNameLock");
    if (on && !mark) { mark = document.createElement("span"); mark.id = "npNameLock"; mark.className = "np-lockmark"; mark.innerHTML = lockSvg; mark.title = "Locked by an admin"; nick.after(mark); }
    if (mark) mark.hidden = !on;
    if (on && document.activeElement !== nick && nick.value !== nameLock) nick.value = nameLock;
  }
  {
    const problem = nameProblem;
    nameProblem = function (raw) {
      const r = problem(raw);
      if (r[1]) return r;
      if (nameLock && signedIn() && r[0] !== nameLock) return [r[0], "Your name was set by an admin and is locked."];
      if (M && M.badName(r[0])) return [r[0], "That name isn’t allowed. Pick a different one."];
      return r;
    };
  }
  if (api) api.listen(data => { nameLock = data && data.nameLock && typeof data.nameLock.name === "string" ? data.nameLock.name : null; renderLock(); });

  // ---------- the jail cell: a banned account sees this instead of the game ----------
  let jail = null, jailedBy = false, jailInfo = null;
  function showJail(info) {
    jailInfo = info || {};
    if (!jail) {
      jail = document.createElement("div"); jail.id = "npJail"; jail.setAttribute("role", "alertdialog"); jail.setAttribute("aria-modal", "true"); jail.setAttribute("aria-labelledby", "npJailTitle");
      jail.innerHTML = '<div class="jw"></div><div class="jl"></div><div class="jb" aria-hidden="true"></div><div class="jsign"><small>Nitro Pitch</small><b id="npJailTitle">You’re banned</b></div>' +
        '<div class="jcard"><div class="why" id="npJailWhy"></div><div class="when" id="npJailWhen"></div><div class="row" id="npJailBtns"></div></div>';
      document.body.append(jail);
      const out = document.createElement("button"); out.type = "button"; out.className = "sbtn"; out.append(h("span", "", "Sign out"));
      out.onclick = async () => { try { await window.npAccounts.logOut(); } catch {} hideJail(true); };

    setTxt("npJailWhy", why ? "Reason: " + why : "The admin banned this account.");
    setTxt("npJailWhen", jailInfo.at ? "Banned on " + new Date(jailInfo.at).toLocaleDateString(undefined, { year: "numeric", month: "long", day: "numeric" }) + ". You can’t play, chat or trade with this account." : "You can’t play, chat or trade with this account.");
    if (!jail.hidden && document.body.classList.contains("np-jailed")) return;
    jail.hidden = false; document.body.classList.add("np-jailed");
    if (!isBanned) { isBanned = true; jailedBy = true; }
    try { if (mode !== "menu") leaveToMenu(true); } catch {}
    try { for (const id of POPUPS) { const el = $(id); if (el) el.hidden = true; } if (screen !== "home") showScreen("home"); } catch {}
  }
  function hideJail(force) {
    if (!jail || jail.hidden) return;
    jail.hidden = true; document.body.classList.remove("np-jailed");
    if (jailedBy || force) { isBanned = false; jailedBy = false; }
  }
  // nothing behind the bars reacts to keys while jailed (Tab still moves between the jail's buttons)
  addEventListener("keydown", e => { if (jail && !jail.hidden && e.key !== "Tab" && !jail.contains(e.target)) { e.stopImmediatePropagation(); e.preventDefault(); } }, true);
  addEventListener("np-banned", e => showJail(e.detail));

  async function checkBan(force) {
    const u = myUid();

    if (!force && u === banUid && Date.now() - banAt < 180000) return;
    banUid = u; banAt = Date.now();
    try { const r = await api.call("ban-status"); if (myUid() !== u) return; if (r && r.banned) showJail(r.banned); else hideJail(); } catch {}
  }
  setInterval(() => { checkBan(false); if (isBanned && !(jail && !jail.hidden) && myUid()) showJail(jailInfo || {}); renderLock(); }, 3000);

    btn.title = unread ? "Messages (" + unread + " unread)" : "Messages";

  function alertNew(c) {
    const title = cleanName(c.name || "A player") + " messaged you";

    if ((document.hidden || !document.hasFocus()) && !pushed && typeof Notification === "function" && Notification.permission === "granted") {
      try { const n = new Notification(title, { body: "Open Nitro Pitch to read and reply.", tag: "np-pm-" + c.with, icon: "/nitro-pitch-icon.png", badge: "/nitro-pitch-icon-48.png" }); n.onclick = () => { window.focus(); n.close(); openThread(c.with, c.name); }; setTimeout(() => n.close(), 30000); } catch {}
    } else if (!(view && view.with === c.with && !pop.hidden)) toast(title);
  }
  let busy = false;
  async function refresh() {
    if (busy || !signedIn() || isBanned) return null;
    busy = true;
    try {
      const r = await opsCall("/api/messages/inbox", { blocks: blockedUids() });
      for (const c of r.convs) { const prev = known.get(c.with); if (!firstInbox && c.unread && !c.mine && (!prev || c.last > prev)) alertNew(c); known.set(c.with, c.last); }

      return r;
    } catch { return null; } finally { busy = false; }

  async function openThread(uid, name, quiet) {

    const msgs = h("div", "pm-msgs");
    if (!t.msgs.length) msgs.append(h("div", "note", "Say hi! Messages are private between you two. Be kind: the chat rules apply here too."));

    const form = document.createElement("form"), input = document.createElement("input"), send = document.createElement("button");
    input.className = "t"; input.maxLength = 300; input.placeholder = blocked ? "You blocked this player" : "Message " + view.name; input.disabled = !!blocked; input.autocomplete = "off";
    send.type = "submit"; send.className = "sbtn pri"; send.append(h("span", "", "Send")); send.disabled = !!blocked;
    form.append(input, send);
    form.onsubmit = async e => {
      e.preventDefault();
      const text = lchatClean(input.value);
      if (!text) return;
      if (muteLeft(myUid()) > 0) { pmNote("You’re muted, so you can’t send messages right now."); return; }
      send.disabled = true;
      try { const r = await opsCall("/api/messages/send", { to: uid, text }); input.value = ""; const d = h("div", "pm-m me", r.sent.tx); d.append(h("small", "", when(r.sent.at))); msgs.querySelector(".note")?.remove(); msgs.append(d); msgs.scrollTop = msgs.scrollHeight; pmNote(""); known.set(uid, r.sent.at); }
      catch (err) { pmNote((err && err.message) || "Couldn’t send that message"); }
      finally { send.disabled = !!blocked; input.focus(); }
    };
    body.append(top, msgs, form, h("div", "note", blocked ? "Unblock them in their player menu to message them." : "Up to 300 characters. Blocked players can’t message you."));
    msgs.scrollTop = msgs.scrollHeight;
    if (!quiet) input.focus();
    refresh();
  }
  // "Message" in a player's stats
  {
    const stats = showStats;
    showStats = function (e) {
      stats.apply(this, arguments);
      const uid = e && (e.uid || (e.pr && e.pr.ui)), me = myUid();
      if (!e || e.me || !uid || !me || uid === me || !UID_RE.test(uid)) return;
      const sec = $("spAdmin"); sec.hidden = false;
      const row = h("div", "row");
      row.append(adBtn("Message", () => { $("statsPop").hidden = true; openThread(uid, e.name); }, true));
      sec.append(h("small", "", "Private message"), row);
    };
  }
  // the service worker reports pushes while a game tab is visible, and notification clicks
  if (navigator.serviceWorker) navigator.serviceWorker.addEventListener("message", e => {
    const d = e.data || {};
    if (d.np === "pm") refresh();
    if (d.np === "open-pm" && typeof d.from === "string") { refresh(); openThread(d.from, ""); }
  });
  if (location.hash === "#messages") { const t = setInterval(() => { if (signedIn()) { clearInterval(t); history.replaceState(null, "", location.pathname + location.search); pop.hidden = false; openInbox(); } }, 1000); setTimeout(() => clearInterval(t), 30000); }
  setInterval(() => {
    const on = signedIn() && !isBanned; btn.hidden = !on;
    if (!on) { setBadge(0); known.clear(); firstInbox = true; if (!pop.hidden) pop.hidden = true; }
  }, 1000);
  setInterval(refresh, 45000);
  setTimeout(refresh, 5000);
  return { refresh, openThread, openInbox, showJail, hideJail, checkBan };
})();

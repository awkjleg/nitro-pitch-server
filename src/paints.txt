// Painted cars from drops. Runs inside the game closure, after secure-progress (which applies server profiles).
// A painted car is its own car body: Titanium White Octane is a different car from the Octane, with the paint on its
// trim (bumpers, skirts, spoiler, mirrors, edges). The car's own colour is still yours to pick in the Paint tab.
// Painted bodies come after the real ones: index BASE + (paint - 1) * BASE + car, the same order as the server's
// catalog (src/paints.js addPaintedCars), so a painted car travels to other players as a plain body number.
const NPPaints = (() => {
  // must match src/paints.js PAINTS (a test checks)
  const PAINTS = [null,
    { name: "Crimson", hex: 0xc8202f, tier: 0 }, { name: "Sky Blue", hex: 0x5fc3f2, tier: 0 }, { name: "Lime", hex: 0x93e23a, tier: 0 },
    { name: "Orange", hex: 0xff8a1e, tier: 0 }, { name: "Grey", hex: 0x8f98a8, tier: 0 }, { name: "Cobalt", hex: 0x2a4fd6, tier: 0 },
    { name: "Forest Green", hex: 0x2f7d3b, tier: 0 }, { name: "Burnt Sienna", hex: 0x9a4a24, tier: 1 }, { name: "Saffron", hex: 0xf2c321, tier: 1 },
    { name: "Pink", hex: 0xff6fb5, tier: 1 }, { name: "Purple", hex: 0x7b3fd1, tier: 2 }, { name: "Gold", hex: 0xd9a520, tier: 3 },
    { name: "Black", hex: 0x15161a, tier: 3 }, { name: "Titanium White", hex: 0xf4f6f8, tier: 4 }];
  const BASE = CARS.length;
  for (let P = 1; P < PAINTS.length; P++) for (let B = 0; B < BASE; B++) {
    const c = CARS[B], admin = c.admin || c.name === "ADMIN" || c.ev;
    // ADMIN / event cars have no painted version: an unobtainable placeholder keeps the numbering the same as the server's
    CARS.push(admin ? { ...c, name: "Unavailable", lv: 999, drop: false, pass: undefined, ev: undefined, admin: undefined, none: true }
      : { ...c, name: PAINTS[P].name + " " + c.name, lv: 999, drop: false, pass: undefined, ev: undefined, admin: undefined, painted: [B, P] });
  }
  // ADMIN design car: Tung Tung Sahur (after the painted cars, same index as the server's catalog). Only the admin
  // has it, and players the admin gives it to (Give items).
  const TUNG = CARS.length;
  CARS.push({ ...CARS[0], name: "Tung Tung Sahur", lv: 1, rar: 7, admin: true, drop: false, pass: undefined, ev: undefined, tung: true });
  const painted = i => { const c = CARS[i]; return c && c.painted ? c.painted : null; };

  // ---- trim paint: a per-colour copy of the car's shared atlas material with the trim cells repainted
  // Atlas cells (CarLib CELL_DEF order, 16 px each in a 512 px atlas): black trim, plastic, carbon, team / teamDark accents.
  const TRIM_CELLS = [[0, 1], [1, 0.85], [6, 0.72], [28, 1], [29, 1], [30, 0.72], [31, 0.72]];
  const trimMats = new Map();
  function trimMat(base, colour) {
    const k = base.uuid + ":" + colour; if (trimMats.has(k)) return trimMats.get(k);
    const src = base.map && base.map.image; if (!src) return base;
    const cv = document.createElement("canvas"); cv.width = src.width; cv.height = src.height;
    const g = cv.getContext("2d"), s = src.width / 512; g.drawImage(src, 0, 0);
    for (const [i, f] of TRIM_CELLS) { g.fillStyle = "#" + new THREE.Color(colour).multiplyScalar(f).getHexString(); g.fillRect((i % 32) * 16 * s, Math.floor(i / 32) * 16 * s, 16 * s, 16 * s); }
    // (a cloned texture shares its image "source" with the original: give the copy its own, or every car's trim changes)
    const t = base.map.clone(); if (THREE.Source) t.source = new THREE.Source(cv); else t.image = cv; t.needsUpdate = true;
    const m = base.clone(); m.map = t; m.name = "carlib-multi-trim"; trimMats.set(k, m); return m;
  }
  const applyTrim = (group, colour) => group.traverse(o => { if (o.isMesh && o.material && o.material.name === "carlib-multi") o.material = trimMat(o.material, colour); });
  // every car the game builds (yours, other players', replays) goes through CarLib.build
  // ---- Tung Tung Sahur: a standing wooden log with a face, thin arms holding a bat, and legs that run as the car drives
  // (the car underneath is an Octane, so it drives and hits exactly like one; only the look changes)
  let TT = null;
  function tungParts() {
    if (TT) return TT;
    const cv = (w, h, draw) => { const c = document.createElement("canvas"); c.width = w; c.height = h; draw(c.getContext("2d"), w, h); const t = new THREE.CanvasTexture(c); t.colorSpace = THREE.SRGBColorSpace; t.anisotropy = 4; return t; };
    const grain = (g, w, h, base) => {
      g.fillStyle = base; g.fillRect(0, 0, w, h);
      for (let i = 0; i < 140; i++) { const x = Math.random() * w; g.strokeStyle = ["#a8642e", "#c98a4b", "#93562a", "#d9a066", "#b7763c"][i % 5]; g.globalAlpha = 0.18 + Math.random() * 0.3; g.lineWidth = 1 + Math.random() * 4; g.beginPath(); g.moveTo(x, 0); for (let y = 0; y <= h; y += 24) g.lineTo(x + Math.sin(y / 60 + i) * 4, y); g.stroke(); }
      g.globalAlpha = 1;
    };
    // the log's skin all the way round, with the face in the middle (u = 0.5 faces forward)
    const body = cv(1024, 512, (g, w, h) => {
      grain(g, w, h, "#c47f42");
      const sh = g.createLinearGradient(0, 0, w, 0); sh.addColorStop(0, "rgba(60,30,10,.35)"); sh.addColorStop(0.5, "rgba(255,220,170,.12)"); sh.addColorStop(1, "rgba(60,30,10,.35)"); g.fillStyle = sh; g.fillRect(0, 0, w, h);
      const cx = 512;
      g.save(); g.translate(cx, 60); g.scale(1.45, 1.12); g.translate(-cx, -60);
      // eyes: big and round, looking slightly to the side
      for (const dx of [-62, 62]) {
        const x = cx + dx, y = 150;
        g.fillStyle = "rgba(70,35,12,.35)"; g.beginPath(); g.ellipse(x, y + 4, 54, 58, 0, 0, Math.PI * 2); g.fill();
        g.fillStyle = "#f7f2ea"; g.beginPath(); g.ellipse(x, y, 46, 50, 0, 0, Math.PI * 2); g.fill();
        g.fillStyle = "#5b3a1e"; g.beginPath(); g.arc(x + 10, y + 4, 25, 0, Math.PI * 2); g.fill();
        g.fillStyle = "#2a1608"; g.beginPath(); g.arc(x + 11, y + 5, 14, 0, Math.PI * 2); g.fill();
        g.fillStyle = "#fff"; g.beginPath(); g.arc(x + 19, y - 5, 7, 0, Math.PI * 2); g.fill();
        g.strokeStyle = "#7a4520"; g.lineWidth = 6; g.beginPath(); g.ellipse(x, y, 47, 51, 0, Math.PI * 1.05, Math.PI * 1.95); g.stroke();
        g.strokeStyle = "#5a3416"; g.lineWidth = 9; g.lineCap = "round"; g.beginPath(); g.moveTo(x - 40, y - 74); g.quadraticCurveTo(x, y - 92, x + 40, y - 76); g.stroke();
      }
      // a long nose with a shadow, then the wide closed-mouth smile and a chin crease
      g.fillStyle = "rgba(70,35,12,.4)"; g.beginPath(); g.moveTo(cx + 6, 170); g.quadraticCurveTo(cx + 26, 250, cx + 22, 282); g.quadraticCurveTo(cx + 4, 292, cx - 6, 284); g.closePath(); g.fill();
      g.fillStyle = "#d0904f"; g.beginPath(); g.moveTo(cx - 10, 168); g.quadraticCurveTo(cx - 26, 248, cx - 20, 278); g.quadraticCurveTo(cx, 296, cx + 20, 278); g.quadraticCurveTo(cx + 18, 230, cx + 8, 168); g.closePath(); g.fill();
      g.fillStyle = "#4a2610"; g.beginPath(); g.ellipse(cx - 11, 280, 6, 4, 0.3, 0, Math.PI * 2); g.ellipse(cx + 11, 280, 6, 4, -0.3, 0, Math.PI * 2); g.fill();
      g.strokeStyle = "#4a2610"; g.lineWidth = 8; g.lineCap = "round"; g.beginPath(); g.moveTo(cx - 74, 318); g.quadraticCurveTo(cx, 368, cx + 78, 314); g.stroke();
      g.strokeStyle = "rgba(255,225,180,.5)"; g.lineWidth = 4; g.beginPath(); g.moveTo(cx - 60, 330); g.quadraticCurveTo(cx, 372, cx + 64, 326); g.stroke();
      g.strokeStyle = "rgba(70,35,12,.45)"; g.lineWidth = 5; for (const dx of [-80, 84]) { g.beginPath(); g.arc(cx + dx, 312, 14, Math.PI * 0.6, Math.PI * 1.4); g.stroke(); }
      g.beginPath(); g.moveTo(cx - 24, 404); g.quadraticCurveTo(cx, 414, cx + 24, 404); g.stroke();
      g.restore();
    });
    const skin = cv(256, 256, (g, w, h) => grain(g, w, h, "#c47f42"));
    const top = cv(256, 256, (g, w, h) => { g.fillStyle = "#dca46a"; g.fillRect(0, 0, w, h); g.strokeStyle = "rgba(120,70,30,.45)"; g.lineWidth = 3; for (let r = 10; r < 128; r += 13) { g.beginPath(); g.arc(128, 128, r, 0, Math.PI * 2); g.stroke(); } });
    TT = {
      body: new THREE.MeshStandardMaterial({ map: body, roughness: 0.78, metalness: 0 }),
      skin: new THREE.MeshStandardMaterial({ map: skin, roughness: 0.8, metalness: 0 }),
      top: new THREE.MeshStandardMaterial({ map: top, roughness: 0.85, metalness: 0 }),
      bat: new THREE.MeshStandardMaterial({ color: 0x8a4f22, roughness: 0.5, metalness: 0 }),
      foot: new THREE.MeshStandardMaterial({ color: 0xc98a52, roughness: 0.75, metalness: 0 })
    };
    return TT;
  }
  function buildTung(o) {
    const car = b0.call(CarLib, 0, o), g = car.g, T = tungParts();
    // hide the Octane: its shell (the meshes before the first wheel pivot) and its wheels; keep the boost flame
    const box = new THREE.Box3();
    for (const c of [...g.children]) { if (!c.isMesh) break; if (c === car.flame) continue; box.expandByObject(c); c.visible = false; }
    for (const w of car.wheels) { const pv = w.userData && w.userData.pivot; (pv || w).traverse(m => { if (m.isMesh) m.visible = false; }); }
    const size = box.getSize(new THREE.Vector3()), mid = box.getCenter(new THREE.Vector3());
    const fwd = car.flame && car.flame.position.z < 0 ? 1 : -1;      // the boost is at the back
    const H = size.z * 0.8, legL = H * 0.34, logH = H * 0.62, r = H * 0.16;
    const man = new THREE.Group(); man.position.set(0, 0, mid.z); man.rotation.y = fwd > 0 ? 0 : Math.PI; g.add(man);
    const add = (geo, mat, x, y, z, parent = man) => { const m = new THREE.Mesh(geo, mat); m.position.set(x, y, z); m.castShadow = true; m.receiveShadow = true; parent.add(m); return m; };
    // the log: slightly narrower at the top, rounded crown with tree rings
    const logGeo = new THREE.CylinderGeometry(r * 0.94, r, logH, 32, 1, true);
    const log = add(logGeo, T.body, 0, legL + logH / 2, 0); log.rotation.y = Math.PI;   // u = 0.5 (the face) toward +z
    // (a sphere cap whose rim sits exactly on the top edge of the log)
    const CAP = Math.PI * 0.3, capR = r * 0.94 / Math.sin(CAP);
    add(new THREE.SphereGeometry(capR, 32, 10, 0, Math.PI * 2, 0, CAP), T.top, 0, legL + logH - capR * Math.cos(CAP), 0);
    add(new THREE.CircleGeometry(r, 28), T.skin, 0, legL, 0).rotation.x = Math.PI / 2;
    // legs (thin, hips at the bottom of the log) with flat bare feet; they swing with the wheels
    const legs = [];
    for (const s of [-1, 1]) {
      const hip = new THREE.Group(); hip.position.set(s * r * 0.42, legL, 0); man.add(hip);
      const leg = add(new THREE.CylinderGeometry(r * 0.12, r * 0.1, legL, 10), T.skin, 0, -legL / 2, 0, hip);
      const foot = add(new THREE.SphereGeometry(r * 0.2, 12, 8), T.foot, 0, -legL + r * 0.06, r * 0.14, hip); foot.scale.set(0.9, 0.42, 1.7);
      legs.push([hip, s]);
    }
    // arms: the left hangs down, the right holds the bat resting on the ground in front
    const shoulderY = legL + logH * 0.55;
    const armL = new THREE.Group(); armL.position.set(-r * 1.02, shoulderY, 0); armL.rotation.z = -0.12; man.add(armL);
    add(new THREE.CylinderGeometry(r * 0.09, r * 0.08, logH * 0.62, 10), T.skin, 0, -logH * 0.31, 0, armL);
    add(new THREE.SphereGeometry(r * 0.13, 10, 8), T.skin, 0, -logH * 0.64, 0, armL);
    const armR = new THREE.Group(); armR.position.set(r * 1.02, shoulderY, 0); armR.rotation.set(-0.5, 0, 0.18); man.add(armR);
    add(new THREE.CylinderGeometry(r * 0.09, r * 0.08, logH * 0.6, 10), T.skin, 0, -logH * 0.3, 0, armR);
    const hand = add(new THREE.SphereGeometry(r * 0.14, 10, 8), T.skin, 0, -logH * 0.62, 0, armR);
    const batLen = H * 0.42, bat = new THREE.Group(); bat.position.copy(hand.position); bat.rotation.set(0.35, 0, -0.25); armR.add(bat);
    add(new THREE.CylinderGeometry(r * 0.2, r * 0.07, batLen, 14), T.bat, 0, -batLen * 0.42, 0, bat);
    add(new THREE.CylinderGeometry(r * 0.1, r * 0.1, r * 0.06, 12), T.bat, 0, batLen * 0.1, 0, bat);
    // run cycle: the legs and arms swing with the wheel spin (the game turns the wheels as the car moves)
    const spin = car.wheels[0];
    log.onBeforeRender = () => {
      const a = spin ? spin.rotation.x : 0, sw = Math.sin(a * 0.9) * 0.55;
      for (const [hip, s] of legs) hip.rotation.x = sw * s;
      armL.rotation.x = -sw * 0.6;
    };
    g.name = "car:Tung Tung Sahur";
    return car;
  }
  const b0 = CarLib.build;
  CarLib.build = function (bi, o) {
    if (CARS[bi] && CARS[bi].tung) return buildTung(o || {});
    const pp = painted(bi); if (!pp) return b0.call(this, bi, o);
    const car = b0.call(this, pp[0], o); applyTrim(car.g, PAINTS[pp[1]].hex); return car;
  };
  // garage and drop tiles (CarLib.thumb builds internally, so painted tiles render here)
  const th0 = CarLib.thumb; let TR = null;
  CarLib.thumb = function (bi, o = {}) {
    const isT = !!(CARS[bi] && CARS[bi].tung);
    if (!painted(bi) && !isT) return th0.call(this, bi, o);
    if (!TR) {
      const cv = document.createElement("canvas"); cv.width = 512; cv.height = 288;
      const r = new THREE.WebGLRenderer({ canvas: cv, antialias: true, alpha: true, preserveDrawingBuffer: true });
      r.outputColorSpace = THREE.SRGBColorSpace; r.toneMapping = THREE.ACESFilmicToneMapping; r.toneMappingExposure = 1.05;
      const sc = new THREE.Scene(); sc.add(new THREE.HemisphereLight(0xcfe0ff, 0x223344, 2.0));
      const d = new THREE.DirectionalLight(0xffffff, 2.8); d.position.set(4, 6, 3); sc.add(d);
      const rl = new THREE.DirectionalLight(0x8fb4ff, 1.6); rl.position.set(-4, 2, -5); sc.add(rl);
      TR = { r, sc, cam: new THREE.PerspectiveCamera(26, 16 / 9, 0.05, 50) };
    }
    const car = CarLib.build(bi, o);
    if (isT) car.g.scale.setScalar(0.72);
    car.g.rotation.y = o.yaw == null ? -0.65 : o.yaw; TR.sc.add(car.g);
    const dist = o.dist || 6.4; TR.cam.position.set(Math.sin(0.35) * dist * 0.3, isT ? 1.0 : 1.25, dist); TR.cam.lookAt(0, isT ? 0.42 : 0.08, 0.05);
    TR.r.setClearColor(0x000000, 0); TR.r.render(TR.sc, TR.cam);
    const out = document.createElement("canvas"); out.width = 256; out.height = 144; out.getContext("2d").drawImage(TR.r.domElement, 0, 0, 256, 144);
    car.dispose();
    return out;
  };

  // ---- painted drops show their paint tier
  const TIER = [["Common", "#9fd0ff"], ["Uncommon", "#78b4ff"], ["Rare", "#b48cff"], ["Very rare", "#ffd451"], ["Rarest", "#ff6fe0"]];
  const odu = openDropUI;
  openDropUI = async function (k) {
    await odu(k);
    const g = dpGot, pp = g && g.kind === "body" && painted(g.i);
    if (pp) setTimeout(() => { if (dpGot !== g) return; const T = TIER[PAINTS[pp[1]].tier]; setTxt("dpRar", "Painted car · " + T[0]); $("dpCard").style.setProperty("--rar", T[1]); }, 1160);
  };
  // car pictures are drawn a couple per frame, normally only in the garage: draw them for the drop reveal too
  (function pump() { try { if (mode === "menu" && !$("dropPop").hidden && thumbs.pending) thumbs.pump(); } catch {} requestAnimationFrame(pump); })();

  // ---- the drops tab no longer counts how many items are left to find
  const rd = renderDrops;
  renderDrops = function (...a) { rd(...a); for (const n of $("npDrops").querySelectorAll(".note")) if (/^(Items still to find in drops|You own every drop item)/.test(n.textContent)) n.remove(); };
  return { PAINTS, BASE };
})();

    msg.textContent = !rows.length ? "Nobody is on the " + NAME() + " leaderboard yet. Play a 2D match while logged in to get on it." : NAME() + " leaderboards: 2D ranks and 2D match stats.";
  };
  // after a 2D ranked match: the rank panel on the 2D ladder (rank.ladder '2d', or a 2D playlist)
  const anim0 = animateRank;
  animateRank = function (rk) {
    if (!rk || !(rk.ladder === "2d" || (rk.pl | 0) >= 4)) return anim0.apply(this, arguments);
    rankAnimT.forEach(clearTimeout); rankAnimT = [];
    const P = RK.PLACEMENTS, bm = rk.before.mmr, am = rk.after.mmr, bp = rk.before.played, ap = rk.after.played, bi = RK.info(bm), ai = RK.info(am);
    const bar = $("prBar"), emb = $("prEmb"), fl = $("prFlash"), e = rk2(rk.pl);
    const at = (ms, f) => rankAnimT.push(setTimeout(f, ms));
    const setBar = (pct, instant) => { if (instant) { bar.style.transition = "none"; bar.style.width = pct + "%"; void bar.offsetWidth; bar.style.transition = ""; } else bar.style.width = pct + "%"; };
    const show = (idx, name, sub) => { const tc = idx >= 0 ? RK.TIER_COLORS[RK.RANKS[idx].tier] : null; emb.src = RK.emblemURL(idx); $("prName").textContent = name; $("prDiv").textContent = sub; $("prName").style.color = tc || "var(--ink)"; $("postRank").style.setProperty("--rkc", (tc || "#8d9bbd") + "66"); };
    const mmrRow = (inf, mmr, placing) => {
      const row = $("prMmr"); row.innerHTML = "";
      const wl = node("span"); wl.append("W ", node("b", "", e[2]), " – L ", node("b", "", e[3]));
      if (placing) { row.append(wl, node("span", "", Math.max(0, P - ap) + " to go")); return; }
      const mid = node("span"); mid.append("MMR ", node("b", "", Math.round(mmr)));
      row.append(node("span", "", inf.min + ""), mid, node("span", "", inf.idx >= RK.TOP ? "" : inf.max + 1 + ""));
    };
    const flash = (text, up) => { fl.className = "pr-flash"; void fl.offsetWidth; fl.textContent = text; fl.className = "pr-flash on " + (up ? "up" : "down"); emb.classList.add("bump"); at(450, () => emb.classList.remove("bump")); };
    $("prTitle").textContent = NAME() + " · Competitive " + (PLAYLISTS[rk.pl] || PLAYLISTS[4]).name;
    $("prDelta").textContent = (rk.d >= 0 ? "+" : "") + rk.d + " MMR"; $("prDelta").style.color = rk.d >= 0 ? "var(--good)" : "var(--danger)";
    fl.className = "pr-flash"; fl.textContent = "";
    const pips = $("prPips"); pips.innerHTML = "";
    if (ap <= P) for (let i = 0; i < P; i++) pips.append(node("i", i < Math.min(e[2], ap) ? "w" : i < Math.min(e[2] + e[3], ap) ? "l" : ""));
    pips.hidden = ap > P;
    const sub = inf => inf.tierName + (inf.lvl ? " · rank " + RK.ROMAN[inf.lvl] : " · top rank");
    if (ap < P) { show(-1, "Unranked", "Placement " + ap + " of " + P); mmrRow(null, 0, true); setBar(bp / P * 100, true); at(500, () => setBar(ap / P * 100)); return; }
    if (bp < P) { show(-1, "Unranked", "Placement " + ap + " of " + P); mmrRow(null, 0, true); setBar(90, true); at(400, () => setBar(100)); at(1300, () => { show(ai.idx, ai.name, sub(ai)); mmrRow(ai, am); setBar(ai.frac * 100, true); flash("Ranked!", true); sfx.levelUp(); }); return; }
    show(bi.idx, bi.name, sub(bi)); mmrRow(bi, bm); setBar(bi.frac * 100, true);
    if (ai.idx === bi.idx) { at(500, () => { setBar(ai.frac * 100); mmrRow(ai, am); }); return; }
    const up = ai.idx > bi.idx;
    at(450, () => setBar(up ? 100 : 0));
    at(1350, () => { show(ai.idx, ai.name, sub(ai)); mmrRow(ai, am); setBar(up ? 0 : 100, true); flash(up ? "Rank up" : "Rank down", up); if (up) sfx.levelUp(); at(120, () => setBar(ai.frac * 100)); });
  };

  renderMenuChrome();
  if (mode === "menu" && screen !== "none") rerenderScreen();
  return { setMenu, is2DMatch, best2, view2, start2D, X_CASUAL, X_RANKED };
})();

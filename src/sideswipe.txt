// Nitro 2D: side-view car soccer (gameplay). Runs inside the game's closure, after the other injected files.
// A 2D match is match.mut[6] === 6. Everything is the normal 3D engine locked to the plane x = 0, which runs
// through both goals: cars and ball keep x = 0, cars only pitch (rotate about x) and face +z or -z, the ball only
// spins about x. Controls, the camera and the bots are swapped for 2D ones while such a match runs; scoring,
// overtime, MVP, stats, XP and quests stay on the game's normal path.
window.NP2D = window.NP2D || {};
(() => {
  const N2 = window.NP2D;
  N2.NAME = "Nitro 2D";                      // the only place the player-facing name is set
  N2.MODE = 6;
  N2.PL = { duel: 4, doubles: 5, standard: 6 };
  N2.MATCH_SECONDS = 120;                    // 2D matches last 2 minutes (mut[0] stays 0)
  const SIZE_PL = { 1: 4, 2: 5, 3: 6 };
  const live = () => mode !== "menu" && (match.mut[6] | 0) === 6;
  N2.on = () => { try { return live(); } catch { return false; } };

  const AIR_SPIN = 3.8, SPIN_T = 0.5, _X = new V3(1, 0, 0), TILT = Math.tan(12 * Math.PI / 180), TURN_T = 0.17, REGEN = 15, KZ = [9.5, 12.5, 14.5], PI = Math.PI;
  const _a = new V3(), _b = new V3(), _f = new V3(), _u = new V3(), _q = new Q(), _vq = new Q(), _Y = new V3(0, 1, 0), _P = new V3();
  const sgn = v => v > 0 ? 1 : v < 0 ? -1 : 0;
  const wrapA = a => { while (a > PI) a -= 2 * PI; while (a < -PI) a += 2 * PI; return a; };
  // which side of the plane the camera watches from: every player attacks to the right of the screen
  const viewSide = () => me.team === 1 ? 1 : -1;                 // camera x sign
  const screenZ = () => me.team === 1 ? -1 : 1;                  // world z of "right" on screen

  // ---------- plane lock ----------
  function lockBall(b) {
    if (!b || !b.p) return;
    b.p.x = 0; if (b.v) b.v.x = 0;
    if (b.w) { b.w.y = 0; b.w.z = 0; }
  }
  function lockCar(c) {
    c.p.x = 0; c.v.x = 0;
    if (c.recT > 0) return;                  // the auto-flip off the roof rolls through the side; snapped when it ends
    _a.set(1, 0, 0).applyQuaternion(c.q);
    const s = Math.abs(_a.x) > 0.05 ? sgn(_a.x) : (c.n2s || 1);
    c.n2s = s;
    if (_a.x * s < 0.999999) { _q.setFromUnitVectors(_a.normalize(), _b.set(s, 0, 0)); c.q.premultiply(_q).normalize(); }
    c.w.y = 0; c.w.z = 0;
  }

  // ---------- the compact 2D arena ----------
  // A 2D match plays in its own small arena on the plane (like Hoops plays on its own court): A2.L either side of
  // centre to the goal lines, a ceiling at A2.H, goals A2.GH tall and A2.GD deep at floor level in both end walls,
  // the end walls rounding into the ceiling (radius RC) and each goal's floor curving up into its back wall (RG).
  // Everything is measured in the (y, z) plane; cars and ball never leave x = 0.
  const A2 = N2.ARENA = Object.freeze({ L: 15, H: 12.5, GH: 5.5, GD: 4.5, RC: 2.5, RG: 1.5 });
  // ...played with bigger cars (CAR_K: hitbox and model) at genre speeds: VMAX on the throttle, VBOOST with boost or in
  // the air (the arena is about 2.5 s across flat out). Gravity, jumps and the ball are the game's.
  const CAR_K = 1.5, VMAX = 11, VBOOST = 14.5;
  N2.CAR_K = CAR_K;
  function scaleCar(c) {
    if (c.n2hb === c.body) return;
    const B = CARS[c.body] || CARS[0];
    c.h = B.h.map(v => v * CAR_K); c.ho = B.ho.map(v => v * CAR_K); c.n2hb = c.body;
  }
  function unscaleCar(c) { if (c.n2hb != null) { setCarBody(c, c.body); c.n2hb = null; } }
  function capSpeed(c, dt) {
    const k = 1 - Math.exp(-7 * dt);
    const m = c.n2cap || 1;
    if (c.ground) { const sp = c.v.length(), cap = (c.boosting ? VBOOST : VMAX) * m; if (sp > cap) c.v.multiplyScalar(1 - k * (1 - cap / sp)); }
    else if (Math.abs(c.v.z) > VBOOST * m) c.v.z -= (c.v.z - sgn(c.v.z) * VBOOST * m) * k;
  }
  const oRemAtt = remAtt;
  remAtt = function (r) { const a = oRemAtt(r); if (live()) { const B = CARS[r.body] || CARS[0]; a.h = B.h.map(v => v * CAR_K); a.ho = B.ho.map(v => v * CAR_K); } return a; };
  function sdf2(p, out) {
    const sz = p.z < 0 ? -1 : 1, az = Math.abs(p.z), y = p.y, AL = A2.L, AH = A2.H, G = A2.GH, RC = A2.RC, RG = A2.RG, back = AL + A2.GD;
    let d = Infinity, ny = 1, nz = 0;
    const take = (dd, a, b) => { if (dd < d) { d = dd; ny = a; nz = b; } };
    if (az <= AL) {
      take(y, 1, 0);
      const cz = AL - RC, cy = AH - RC, round = az > cz && y > cy;
      if (round) { const a = az - cz, b = y - cy, l = Math.hypot(a, b) || 1e-6; take(RC - l, -b / l, -a / l); }
      else take(AH - y, -1, 0);
      if (y >= G) { if (!round) take(AL - az, 0, -1); }
      else { const a = AL - az, b = G - y, l = Math.hypot(a, b) || 1e-6; take(l, -b / l, -a / l); }   // the crossbar's edge
    } else if (az <= back && y < G) {
      take(y, 1, 0); take(G - y, -1, 0);
      const bz = back - RG;
      if (az > bz && y < RG) { const a = az - bz, b = RG - y, l = Math.hypot(a, b) || 1e-6; take(RG - l, b / l, -a / l); }
      else if (az > back - 1 && y > G - 1) { const a = az - (back - 1), b = y - (G - 1), l = Math.hypot(a, b) || 1e-6; take(1 - l, -b / l, -a / l); }
      else take(back - az, 0, -1);
    } else if (y >= G && az <= back + 2) {
      // inside the wall above a goal: out the short way (into the field or down into the goal)
      const p1 = az - AL, p2 = y - G;
      if (p1 < p2) take(-p1, 0, -1); else take(-p2, -1, 0);
    } else take(back - az, 0, -1);                                            // behind a goal
    out.set(0, ny, nz * sz);
    return d;
  }
  const oSDF = arenaSDF;
  arenaSDF = function (p, out, pad = 0) { return live() ? sdf2(p, out) : oSDF(p, out, pad); };
  // goals: the ball fully over a compact goal line scores. The game's check looks at the full-size line, so the
  // host's match logic sees the ball there for that one check (scoring, assists, stats and MVP stay the game's).
  const SCORE_Z = L + 4.25 * UU + 0.5;
  const oAuth = authorityMatch;
  authorityMatch = function (dt) {
    if (!(live() && match.ph === "play" && Math.abs(ball.p.z) > A2.L + BR + 0.05)) return oAuth(dt);
    const z = ball.p.z; ball.p.z = sgn(z) * (SCORE_Z + BR);
    try { return oAuth(dt); } finally { ball.p.z = z; }
  };
  const oPredGoal = predictGoal;
  predictGoal = function (b) {
    if (!live()) return oPredGoal(b);
    const t = { p: b.p.clone(), v: b.v.clone(), w: (b.w || new V3()).clone() }, keep = bounceImpact;
    for (let i = 0; i < 360; i++) { stepBall(t, STEP); if (Math.abs(t.p.z) > A2.L + BR) { bounceImpact = keep; return t.p.z > 0 ? 1 : 0; } }
    bounceImpact = keep;
    return -1;
  };
  const oBotPredict = botPredict;
  botPredict = function () {
    if (!live()) return oBotPredict();
    const keep = bounceImpact, s = { p: ball.p.clone(), v: ball.v.clone(), w: (ball.w || new V3()).clone() };
    BP.gi = -1; BP.gs = 0;
    for (let i = 0; i < BP_N; i++) {
      if (BP.gi < 0) { stepBall(s, BP_DT); if (Math.abs(s.p.z) > A2.L + BR) { BP.gi = i; BP.gs = sgn(s.p.z); } }
      BP.x[i] = s.p.x; BP.y[i] = s.p.y; BP.z[i] = s.p.z;
    }
    bounceImpact = keep; BP.t0 = botClock; BP.dirty = false;
  };
  // the goal explosion and its blast go off at the compact goal
  const atGoal = at => { if (live() && at && Math.abs(at.z) > A2.L + A2.GD + 4) { at = at.clone(); at.z = sgn(at.z) * (A2.L + 1.5); at.y = Math.min(at.y, A2.GH - 1); } return at; };
  const oFxGoal = fx.goal;
  fx.goal = function (type, at, hx) { return oFxGoal.call(this, type, atGoal(at), hx); };
  const oBlast = goalBlast;
  goalBlast = function (c, at) { return oBlast(c, atGoal(at)); };

  // ---------- controls: a 2D stick (world z, y) -> the engine's inputs ----------
  // ground: the stick picks the way to drive along the surface; the other way turns the car around (a quick 180 about
  // the surface normal, momentum kept, then the brakes). Holding the same direction keeps driving (up a wall, onto the
  // ceiling). air: left/right spins the car (clockwise on screen for right); jump with left/right flips that way,
  // jump without them is a double jump.
  function turnAround(c) {
    _q.setFromAxisAngle(c.n, PI);
    c.q.premultiply(_q).normalize(); c.pq.premultiply(_q).normalize();
    // the drawn car swings round over TURN_T with its nose passing the camera side
    _a.set(0, 0, 1).applyQuaternion(_vq.setFromAxisAngle(_Y, -PI / 2)).applyQuaternion(c.q);
    c.n2sg = _a.x * viewSide() > 0 ? 1 : -1;
    c.n2t = TURN_T;
  }
  // keyboard (and a gamepad's triggers): forward / back along the car. W drives the way the car faces; S brakes, then
  // reverses; S pressed at (near) standstill turns the car round, and holding it then drives off the new way. In the
  // air W / S spin the car (nose down / up), and pick the flip at a jump (none: double jump).
  function mapKeys(c, out, kb) {
    _f.set(0, 0, 1).applyQuaternion(c.q);
    if (!c.ground) { c.n2k = 0; out.pitch = kb; out.dx = kb; return out; }
    if (kb > 0) { c.n2k = 0; out.thr = kb; return out; }
    if (kb === 0) { c.n2k = 0; return out; }
    if (!c.n2k) c.n2k = c.v.dot(_f) > 2.5 ? 1 : 2;          // 1: braking hold; 2: a fresh press from a stop
    if (c.n2k === 2) { turnAround(c); c.n2k = 3; }
    out.thr = c.n2k === 3 ? -kb : kb;                        // after the turn: drive off; otherwise brake / reverse
    return out;
  }
  function mapInput(c, inp) {
    const raw = inp.n2;
    const out = { ...inp, thr: 0, steer: 0, roll: 0, ar: false, hb: false, pitch: 0, dx: 0 };
    if (!raw) { c.n2l = null; return out; }
    if (raw.kb !== undefined) return mapKeys(c, out, clamp(+raw.kb || 0, -1, 1));
    const wz = clamp(+raw.z || 0, -1, 1), wy = clamp(+raw.y || 0, -1, 1), mag = Math.min(1, Math.hypot(wz, wy));
    _f.set(0, 0, 1).applyQuaternion(c.q);
    const s = c.n2s || 1;
    if (c.ground) {
      if (mag < 0.25) { c.n2l = null; if (raw.trig) out.thr = clamp(raw.trig, -1, 1); return out; }
      let want = 1;
      const l = c.n2l;
      if (!(l && l.z * wz + l.y * wy > 0.5 * mag)) {
        const n = c.n;
        let along = _f.z * wz + _f.y * wy;
        if (Math.abs(n.y) < 0.75) along -= (n.z * wz + n.y * wy) * (_f.y >= 0 ? 1 : -1);   // pushing into a wall climbs it
        want = along > 0.2 ? 1 : along < -0.2 ? -1 : 0;
        c.n2l = want ? { z: wz / mag, y: wy / mag } : null;
      }
      if (want < 0) turnAround(c);
      if (want) out.thr = mag > 0.85 ? 1 : mag;
      else if (raw.trig) out.thr = clamp(raw.trig, -1, 1);
      return out;
    }
    c.n2l = null;
    out.pitch = clamp(wz * s, -1, 1);
    const flipping = c.flipX && c.flipAge < FLIP_TORQUE_TIME + 0.3;
    if (flipping) out.dx = Math.abs(wz) >= 0.5 && sgn(wz) !== c.n2fd ? -sgn(c.flipX) : 0;   // against the flip: cancel it
    else {
      const fz = Math.abs(_f.z) > 1e-3 ? sgn(_f.z) : 1;            // (the engine's dodge uses the flattened nose)
      out.dx = Math.abs(wz) >= 0.5 ? sgn(wz) * fz : 0;
      c.n2fdp = sgn(wz);
    }
    return out;
  }

  const oStepCar = stepCar;
  stepCar = function (c, inp, dt, jp) {
    if (!live()) { unscaleCar(c); return oStepCar(c, inp, dt, jp); }
    scaleCar(c);
    lockCar(c);
    const m = mapInput(c, inp || {});
    // on the roof: rolls back over by itself after a moment (jump does it at once)
    if (c.ground === false && c.restT > 0 && carUp(c, _u).dot(c.restN) < -0.45) { c.n2tt = (c.n2tt || 0) + dt; if (c.n2tt > 0.6) { jp = true; c.n2tt = 0; } }
    else c.n2tt = 0;
    const wasDodged = c.dodged, wasGround = c.ground;
    oStepCar(c, m, dt, jp);
    if (c.flipAge === 0 && c.flipX) c.n2fd = c.n2fdp || sgn(c.v.z);
    // a double jump spins the car right round (drawn), the way it is pitching or else nose first
    if (!wasDodged && c.dodged && !wasGround && !c.ground && c.flipAge > 0) { c.n2spin = 1; c.n2spd = sgn(c.w.x) || 1; }
    // held air spin: faster than a 3D car can turn, so it keeps rotating like a pinwheel
    if (!c.ground && Math.abs(m.pitch) > 0.9 && !(c.flipX && c.flipAge < FLIP_TORQUE_TIME + 0.3) && !(c.recT > 0)) {
      c.q.multiply(_vq.setFromAxisAngle(_X, sgn(m.pitch) * AIR_SPIN * dt)).normalize();
    }
    capSpeed(c, dt);
    lockCar(c);
  };
  const oStepBall = stepBall;
  stepBall = function (b, dt) {
    if (!live()) return oStepBall(b, dt);
    lockBall(b); oStepBall(b, dt); lockBall(b);
  };
  const oCarBall = carBall;
  const hitQ = [];
  carBall = function (c, b) {
    const vy = b.v.y, vz = b.v.z;
    const r = oCarBall(c, b);
    if (r && live()) {
      lockBall(b); c.v.x = 0; c.p.x = 0;
      // a bot's touch is softer the lower its rank (n2pow, set by the bot brain; players always hit at full power)
      if (c.n2pow && c.n2pow < 1) { b.v.y = vy + (b.v.y - vy) * c.n2pow; b.v.z = vz + (b.v.z - vz) * c.n2pow; }
      const dv = Math.hypot(b.v.y - vy, b.v.z - vz);
      if (b === ball && dv > 12 && hitQ.length < 4 && nowS() - (ball.n2hit || -9) > 0.12) { ball.n2hit = nowS(); hitQ.push({ p: b.p.clone(), s: dv, col: c === me.car ? TEAM_COL[me.team] || 0xffffff : 0xffffff }); }
    }
    return r;
  };
  function teamOfCar(c) {
    if (c === me.car) return me.team;
    for (const b of bots) if (b.car === c) return b.team;
    for (const [, r] of remotes) if (r.att === c) return r.team;
    return -1;
  }
  // teammates share the lane: they drive through each other (opponents still bump)
  const oCarCar = carCar;
  carCar = function (c, a, key) {
    if (!live()) return oCarCar(c, a, key);
    const tc = teamOfCar(c);
    if (tc >= 0 && tc === teamOfCar(a)) return false;
    const r = oCarCar(c, a, key);
    c.p.x = 0; c.v.x = 0;
    return r;
  };
  // no demolitions in a single lane
  const oDemo = demoCheck;
  demoCheck = function (c, t) { return live() ? false : oDemo(c, t); };
  // the drawn car swings round during a turn-around
  const oInterp = interpCar;
  interpCar = function (c, a, dt) {
    oInterp(c, a, dt);
    if (c.n2t > 0) {
      c.n2t = Math.max(0, c.n2t - dt);
      if (!live()) { c.n2t = 0; return; }
      const k = c.n2t / TURN_T, e = k * k * (3 - 2 * k);
      c.rq.multiply(_vq.setFromAxisAngle(_Y, -c.n2sg * PI * e));
    }
    if (c.n2spin > 0) {
      c.n2spin = Math.max(0, c.n2spin - dt / SPIN_T);
      if (!live() || c.ground) { c.n2spin = 0; return; }
      const k = 1 - c.n2spin, e = 1 - (1 - k) * (1 - k) * (1 - k);             // fast, easing out
      c.rq.multiply(_vq.setFromAxisAngle(_X, c.n2spd * 2 * PI * e));
    }
  };
  // online: remote cars come from clients that lock their own car; keep the drawn and predicted spots on the plane too
  const oSnap = sampleSnap;
  sampleSnap = function (r, nowMs) {
    oSnap(r, nowMs);
    if (live()) { if (r.p) r.p.x = 0; if (r.cp) r.cp.x = 0; }
  };
  // the host's match state: the game clamps mut[6] to 0..5 and pl to 0..3; keep 2D matches 2D on every client
  const oHost = applyHostState;
  applyHostState = function (g) {
    oHost(g);
    if (!g || typeof g !== "object") return;
    if (Array.isArray(g.mut) && g.mut.length >= 7 && (g.mut[6] | 0) === 6) match.mut[6] = 6;
    if (g.pl === 4 || g.pl === 5 || g.pl === 6) match.pl = g.pl;
  };
  // 2-minute matches
  const oFresh = freshMatch;
  freshMatch = function (base) {

  };

  // ---------- spawns ----------
  const oKick = kickSpot;
  kickSpot = function (team, idx, k) {

    let slot = idx;
    if (n >= 2 && idx < n) {

      const order = [...Array(n).keys()], r = rng((k | 0) * 7919 + 29);
      for (let i = n - 1; i > 0; i--) { const j = Math.floor(r() * (i + 1)); [order[i], order[j]] = [order[j], order[i]]; }

  };

    for (const c of cars) { if (boostRule() === 0) c.boost = 100; c.n2l = null; c.n2t = 0; c.n2s = 0; lockCar(c); c.pq.copy(c.q); c.rq.copy(c.q); }
  };

  // ---------- input ----------
  const oRead = readInput;
  readInput = function () {
    const r = oRead();
    const ap = N2.test && N2.test.pilot;
    if (ap && live() && (me.team === 0 || me.team === 1)) {
      if (ap.car !== me.car || ap.team !== me.team) { ap.car = me.car; ap.team = me.team; ap.n2 = null; }
      const busy = !inGamePhase(match.ph) || match.ph === "countdown" || match.ph === "replay";
      return busy ? r : { ...r, ...think2D(ap, 1 / 60) };
    }
    if (live()) {
      // the phone joystick and a gamepad's left stick are screen-space (left / right on screen, up / down for air
      // spin); keyboard keys and gamepad triggers drive along the car. A / D (steer) do nothing in 2D.
      let sx = 0, sy = 0, kb = (held("fwd") ? 1 : 0) - (held("back") ? 1 : 0);
      const pads = padList(), B = settings.binds;
      if (touch.x || touch.y) { sx = touch.x; sy = -touch.y; }
      else if (pads.length) {
        const ax = k => pads.reduce((m, gp) => Math.abs(gp.axes[k] || 0) > Math.abs(m) ? gp.axes[k] : m, 0);
        const x = ax(0), y = ax(1);
        if (Math.abs(x) > settings.dead) sx = x;
        if (Math.abs(y) > settings.dead) sy = -y;
        const val = i => i < 0 ? 0 : Math.max(0, ...pads.map(gp => gp.buttons[i] ? gp.buttons[i].value : 0));
        const trig = val(B.fwd.p) - val(B.back.p);
        if (Math.abs(trig) > 0.05) kb = trig;
      }
      r.n2 = Math.hypot(sx, sy) > 0.05 ? { z: clamp(sx, -1, 1) * screenZ(), y: clamp(sy, -1, 1) } : { kb };
    }
    return r;
  };
  const oBallCam = toggleBallCam;
  toggleBallCam = function () { if (live()) return; oBallCam(); };

  // ---------- per frame: boost refill, labels, UI ----------
  let lastMid = "", bodyOn = false;

    if (!live()) return;
    if (match.mid && match.mid !== lastMid) { lastMid = match.mid; cam.snap = true; if (me.team === 0 || me.team === 1) toast(N2.NAME + ": score in the goal on the right! Jump, flip, spin and boost."); }
    // boost refills by itself (the lane passes few pads)

      if (me.team === 0 || me.team === 1) list.push(me.car);
      if (isAuthority) for (const b of bots) list.push(b.car);

  // ---------- side camera ----------
  const cam = { f: new V3(), d: 40, w: 30, snap: true };
  // Side view from slightly above (TILT, ~12 degrees down) with a normal lens: the pitch reads as a strip in front of
  // the stadium. Framed by the width of view at the lane: in landscape most of the compact arena (goal to goal with
  // gentle panning, zooming out to all of it when the play is spread); on a portrait phone a tighter window that pans.
  function sideCamera(dt) {
    if (viewOffsetOn) { camera.clearViewOffset(); viewOffsetOn = false; camera.updateProjectionMatrix(); }
    dt = clamp(dt, 0, 0.1);
    const a = innerWidth / Math.max(1, innerHeight), portrait = a < 1;
    const B = ballMesh.position;
    const player = match.ph !== "replay" && !spec && (me.team === 0 || me.team === 1) && !me.car.dead;
    const C = player ? me.car.rp : B;
    const fz = player ? C.z + (B.z - C.z) * 0.55 : B.z, fy = player ? C.y + (B.y - C.y) * 0.5 : B.y;
    const full = 2 * (A2.L + A2.GD) + 3;
    const base = portrait ? 18 : full, most = portrait ? 26 : full;
    const mz = portrait ? 4 : 5;
    const needW = 2 * (Math.max(Math.abs(B.z - fz), Math.abs(C.z - fz)) + mz);
    const want = clamp(Math.max(base, needW), base, most);
    if (cam.snap) { cam.w = want; cam.f.set(0, fy, fz); }
    cam.w += (want - cam.w) * (1 - Math.exp(-(want > cam.w ? 3 : 1.2) * dt));
    cam.f.z += (fz - cam.f.z) * (1 - Math.exp(-5 * dt));
    cam.f.y += (fy - cam.f.y) * (1 - Math.exp(-4 * dt));
    cam.snap = false;
    const vfov = portrait ? 58 : 34, hz = cam.w / 2, hy = hz / a;
    const D = hz / (Math.tan(vfov * PI / 360) * a);
    cam.d = D;
    let y = cam.f.y, z = cam.f.z;
    if (player) { const kz = Math.min(hz * 0.7, mz); z = clamp(z, C.z - hz + kz, C.z + hz - kz); cam.f.z = z; }
    // the floor about a fifth of the way up (the HUD lives along the bottom); in portrait about a third (above the
    // touch controls), the whole arena height in view and the stands above it
    const lo = hy * (portrait ? 0.36 : 0.62), hi = A2.H + 3 - hy * 0.85;
    y = lo > hi ? lo : clamp(y, lo, hi);
    const zMax = Math.max(0, A2.L + A2.GD + 1 - hz);
    z = clamp(z, -zMax, zMax);
    let sy = 0, sz = 0;
    if (shakeT > 0) { shakeT = Math.max(0, shakeT - dt); if (settings.shake && !NPDrive.phone()) { sy = (Math.random() - 0.5) * shakeT * 0.5; sz = (Math.random() - 0.5) * shakeT * 0.5; } }
    camera.up.set(0, 1, 0);
    camera.position.set(viewSide() * D, y + D * TILT + sy, z + sz);
    camLook.set(0, y, z + sz);
    camera.lookAt(camLook);
    camPos.copy(camera.position); camUp.set(0, 1, 0); camFollow = false;
    if (Math.abs(camera.fov - vfov) > 0.01) { camera.fov = vfov; camera.updateProjectionMatrix(); }
  }

  // ---------- bots ----------
  // 2D bots press the same 2D stick (world z / y), jump and boost that players do; skill sk is the game's bot skill
  // (match skill from botMatchSkill / the rank-based bm, plus a personal offset): t = 0 at Bronze, 1 at SSL.
  //   low: slow reactions (0.4 s), sloppy approach points, ball chasing (double commits), mistimed jumps, boost spent
  //        early at kickoffs, no aerials
  //   mid: quicker reads, rotation (attacker, support behind the play, keeper in 3v3), jump shots and saves, now and
  //        then a power flip into an uncontested ball
  //   high: 0.12 s reads, aerial saves (Diamond and up), aerial shots in their half (Champion and up, double-jump
  //        aerials at the top), boost kept for the air
  // Tuned with bot-vs-bot batches (tests' NP2D.test hooks): strength rises with skill at every step. b.flags (tests
  // only) switches single behaviours off for such comparisons.
  const BOT_IN = () => ({ thr: 0, steer: 0, boost: false, hb: false, roll: 0, jp: false, jh: false, n2: { z: 0, y: 0, trig: 0 } });
  const fwdA = c => { _f.set(0, 0, 1).applyQuaternion(c.q); return Math.atan2(_f.y, _f.z); };
  const upA = c => { carUp(c, _u); return Math.atan2(_u.y, _u.z); };
  const spinX = c => _a.copy(c.w).applyQuaternion(c.q).x;
  // a stick value that turns an in-plane angle (now) to target: a PD controller on the spin about x
  function spinTo(c, now, target) { const e = wrapA(target - now); return clamp(-3.6 * e - 0.9 * spinX(c), -1, 1); }
  function facing(c) { _f.set(0, 0, 1).applyQuaternion(c.q); return sgn(_f.z) || 1; }
  // drive toward z (rush: full speed through it); otherwise brake to stop on it, then face o.face
  function driveTo(b, c, tz, inp, o) {
    tz = clamp(tz, -A2.L + 2.5, A2.L - 2.5);
    const dz = tz - c.p.z, vz = c.v.z, sp = Math.abs(vz), st = b.n2;
    const toward = sgn(dz) === sgn(vz), f = facing(c);
    const stop = sp * sp / (2 * 55) + 0.6;
    if (Math.abs(dz) < 1.6 && sp < 5) {
      if (o.face && f !== o.face && c.ground && st.faceCD <= 0) { inp.n2.z = o.face; st.faceCD = 0.6; }
      return;
    }
    // arriving: brake (the reverse pedal) instead of turning round; short moves backwards are reversed into
    if (!o.rush && toward && Math.abs(dz) < stop) { if (c.ground) inp.n2.trig = sgn(vz) === f ? -1 : 1; return; }
    if (!o.rush && c.ground && f !== sgn(dz) && Math.abs(dz) < 7 && sp < 9) { inp.n2.trig = -1; return; }
    inp.n2.z = sgn(dz) * (o.rush && st.save ? 1 : st.drive || 1);
    const okDir = c.ground ? facing(c) === sgn(dz) && c.n.y > 0.7 : false;
    if (okDir && (o.rush ? Math.abs(dz) > (st.lateBoost || 0) : Math.abs(dz) > o.boostD + (st.lateBoost || 0)) && sp < VBOOST - 0.5 && (c.boost > o.keep + (st.hoard || 0) || boostRule() === 1)) inp.boost = true;
  }
  function stState(b) {
    return b.n2 || (b.n2 = { think: 0, role: 0, act: null, faceCD: 0, noise: 0, noiseT: 0, plan: null, chase: false });
  }
  // where to meet the ball: first point on the predicted path this car can reach from its own goal side
  const F2 = b => b.flags || NOFLAGS;
  function intercept(b, c, t, atk, urgent) {
    const age = botClock - BP.t0, lim = BP.gi >= 0 ? BP.gi : BP_N;
    const bst = c.boost > 12 || boostRule() === 1, acc = bst ? 30 : 18, vm = bst ? VBOOST : VMAX;
    const maxAir = t < 0.55 || (b.flags && b.flags.noAerial) ? 0 : Math.min(A2.H - 2, 8 + t * 22);   // Diamond and up
    let air = null;
    for (let i = 1; i < lim; i += 2) {
      const T = (i + 1) * BP_DT - age; if (T < 0.05) continue;
      const pz = BP.z[i], py = BP.y[i];
      const az = pz - atk * (BR + 1.1 * CAR_K), d = Math.abs(az - c.p.z), dir = sgn(az - c.p.z) || atk;
      if ((c.p.z - pz) * atk > 0.5) continue;                       // the ball is behind us
      let vT = c.v.z * dir, tt = vT < -1 ? -vT / 70 + 0.12 : 0;
      vT = Math.max(0, vT);
      const t1 = Math.max(0, (vm - vT) / acc), d1 = (vT + vm) / 2 * t1;
      tt += d <= d1 ? (-vT + Math.sqrt(vT * vT + 2 * acc * d)) / acc : t1 + (d - d1) / vm;
      let kind;
      if (py < 3.3) kind = 0;
      else if (py < 6.3 && t >= (F2(b).jumpT ?? 0.25) && botRise(py - 1.2) > 0) { kind = 1; tt = Math.max(tt, botRise(py - 1.2) + 0.05); }
      else if (py < maxAir && c.boost > 18 + py) { kind = 2; tt = Math.max(tt, (py - 1) / 13 + 0.35); }
      else continue;
      if (tt > T) continue;
      const hit = { T, kind, z: pz, y: py, at: botClock + T };
      if (kind < 2) return air && urgent && air.T < T - 0.4 ? air : hit;
      // aerials: saves, and (Champion and up) shots in their half that nobody else can reach first
      if (!air && (urgent || (t >= (F2(b).airT ?? 0.75) && pz * atk > 0 && !b.n2.press))) air = hit;
      if (air && T > air.T + 1.2) break;
    }
    return air;
  }
  // a decision from what the bot sees now; think2D acts on it after the bot's reaction time
  function decide(b, c, t, atk) {
    const S = stState(b), st = {}, team = BT[b.team], ownZ = -atk * A2.L;
    const kick = ball.v.lengthSq() < 0.05 && Math.abs(ball.p.z) < 0.05;
    // rank the team by who should play the ball: distance, plus a penalty for being upfield of it
    const score = o => kick ? Math.abs(o.p.z) : Math.abs(ball.p.z - o.p.z) / 30 + ((o.p.z - ball.p.z) * atk > 1 ? 1.1 : 0) + (o.p.y > 4 ? 0.2 : 0);
    const mine = score(c);
    if (!(S.chaseT > botClock)) { S.chaseT = botClock + 1; S.chaseOn = Math.random() < (F2(b).chase ?? (1 - t) * 0.45); }
    st.chase = !kick && S.chaseOn;
    let rank = 0;
    if (!st.chase) for (const o of team) if (o.p !== c.p) { const s = score(o); if (s < mine - 1e-6 || (Math.abs(s - mine) < 1e-6 && o.b && o.b.key < b.key)) rank++; }
    st.role = rank; st.kick = kick;
    const tGoal = BP.gs === -atk && BP.gi >= 0 ? (BP.gi + 1) * BP_DT - (botClock - BP.t0) : 99;
    let closest = true;
    for (const o of team) if (o.p !== c.p && Math.abs(o.p.z - ownZ) < Math.abs(c.p.z - ownZ)) closest = false;
    // saves from Platinum up; higher ranks read the shot earlier (Platinum ~1.2 s before it would go in, Legend 3 s)
    st.save = t > (F2(b).saveT ?? 0.5) && tGoal < bmix(1.2, 3, clamp((t - 0.5) / 0.5, 0, 1)) && !F2(b).noSave && (rank === 0 || closest);
    // a challenge: an opponent is about to play the ball too
    const eta = o => Math.abs(ball.p.z - o.p.z) / Math.max(12, Math.abs(o.v.z) * 0.6 + 18);
    let oppEta = 9; for (const o of BT[1 - b.team]) if ((o.p.z - ball.p.z) * atk >= -1) oppEta = Math.min(oppEta, eta(o));
    st.press = oppEta < eta(c) + 0.35;
    // shadow defence (Gold and up, more often higher): the opponent will clearly get there first and the ball is in
    // our half: hold a spot in front of our goal and step up only when the ball comes close
    st.shadow = !kick && rank === 0 && closest && oppEta + 0.25 < eta(c) && ball.p.z * atk < 4 && t >= (F2(b).shadowT ?? 0.35) && Math.random() < bmix(0.5, 1, t);
    st.plan = !kick && (rank === 0 || st.save) ? intercept(b, c, t, atk, tGoal < 3) : null;
    // the random handicaps are drawn on a clock, not per decision (so quicker thinking never means more mistakes):
    // an aim offset, and now and then a hesitation (low ranks pull up short of the ball)
    const tn = F2(b).noiseT ?? Math.pow(t, 1.4);
    if (!(S.mistakeT > botClock)) {
      S.mistakeT = botClock + 0.9;
      S.noise = (Math.random() - 0.5) * (1 - tn) * 3;
      S.wrongOn = Math.random() < (F2(b).wrongP ?? (1 - tn) * 0.35);
      S.placeOn = Math.random() < (F2(b).placeP ?? t);
    }
    st.wrong = !kick && S.wrongOn; st.placeOn = S.placeOn;
    return st;
  }
  function startAct(st, kind, o) { st.act = { kind, t: 0, ...o }; }
  // a move in progress owns the controls; false when it is over
  function runAct(b, c, t, atk, dt, inp) {
    const st = b.n2, A = st.act; A.t += dt;
    const toBall = sgn(ball.p.z - c.p.z) || atk;
    if (A.kind === "flip") {
      // jump, then flip toward dir
      if (A.t < 0.07) { inp.jh = true; inp.n2.z = 0; if (A.t <= dt) inp.jp = true; return true; }
      if (!A.done) { if (c.ground) return false; inp.jp = true; inp.n2.z = A.dir; A.done = true; return true; }
      if (A.t < 0.75) { inp.n2.z = A.dir; return !c.ground || A.t < 0.2; }
      return false;
    }
    if (A.kind === "jump") {
      // jump shot: hold the jump, point the nose at the ball, flip into it when it is close
      if (A.t <= dt) { inp.jp = true; inp.jh = true; return true; }
      if (c.ground && A.t > 0.15) return false;
      inp.jh = A.t < 0.22;
      _P.subVectors(ball.p, c.p);
      if (!A.flip && _P.length() < 3.4 + BR && t > 0.25 && Math.abs(_P.z) > 0.6) { A.flip = true; inp.jp = true; inp.n2.z = sgn(_P.z); return true; }
      if (A.flip) { inp.n2.z = sgn(_P.z) || toBall; return A.t < 1.2; }
      inp.n2.z = spinTo(c, fwdA(c), Math.atan2(_P.y, _P.z));
      return A.t < 1.3;
    }
    if (A.kind === "aerial") {
      if (A.t <= dt) { inp.jp = true; inp.jh = true; return true; }
      if (c.ground && A.t > 0.2) return false;
      inp.jh = A.t < 0.2;
      if (A.dbl && A.t > 0.17 && !A.dj) { A.dj = true; inp.jp = true; inp.n2.z = 0; return true; }
      const left = Math.max(0.05, A.at - botClock);
      botBallAt(left, _P);
      _P.z -= atk * BR * 0.8; _P.y -= 0.3;                            // a little behind and under: it goes their way
      _P.sub(c.p);
      const want = Math.atan2(_P.y - 0.5 * CAR_G0 * left * left * 0.4, _P.z), now = fwdA(c);
      inp.n2.z = spinTo(c, now, want);
      if (Math.abs(wrapA(want - now)) < 0.55 && c.boost > 0) inp.boost = true;
      const d = _P.length();
      if (A.t > 0.45 && A.last != null && d > A.last + 0.05 && !A.flip) return false;   // going past it: give up
      A.last = d;
      if (!A.flip && d < 3 + BR && t > 0.7 && A.t > 0.3) { A.flip = true; inp.jp = true; inp.n2.z = sgn(_P.z) || atk; }
      return A.t < A.T + 0.6 && (d < 40 || A.t < 0.6);
    }
    return false;
  }
  // bot skill from a 2D MMR: the 2D ladder's tiers line up with the 3D ones (2D Bronze = 3D Bronze ... 2D Champion
  // spans Champion and Grand Champion, Legend = Supersonic Legend and up) and go through the game's own curve
  const MAP2D = [[0, 0], [275, 294], [550, 475], [825, 655], [1100, 835], [1375, 1075], [1650, 1861], [1850, 2000]];
  N2.mmrTo3D = function (m) {
    m = fin(+m) ? +m : Ranked.START;
    if (m <= 0) return 0;
    for (let i = 1; i < MAP2D.length; i++) if (m <= MAP2D[i][0]) { const [a0, b0] = MAP2D[i - 1], [a1, b1] = MAP2D[i]; return b0 + (b1 - b0) * (m - a0) / (a1 - a0); }
    return MAP2D[MAP2D.length - 1][1] + (m - MAP2D[MAP2D.length - 1][0]);
  };
  N2.botSkill = m => Ranked.botSkill(N2.mmrTo3D(m));
  // the match's bot skill: matchmade games (ranked and casual, bot-filled) from the bots' MMR bm (the humans' 2D MMR),
  // exhibitions from the chosen difficulty
  N2.matchSkill = () => match.mm && match.bm > 0 ? N2.botSkill(match.bm) : botMatchSkill();
  const NOFLAGS = {};
  function think2D(b, dt) {
    const c = b.car, atk = b.team === 0 ? 1 : -1, ownZ = -atk * A2.L;
    const sk = b.sk = fin(b.skx) ? b.skx : clamp(N2.matchSkill() + (b.skd || 0), 0.3, 1.25);
    const t = clamp((sk - 0.45) / 0.67, 0, 1);
    const inp = BOT_IN();
    if (c.dead) return inp;
    const st = stState(b), F = b.flags || NOFLAGS;
    st.faceCD -= dt;
    // handicaps that fall away with skill: a lighter foot (Bronze drives at ~70% throttle) and boost hoarded unused
    // (the handicaps ease off slowly through the low and middle ranks and fast towards the top: u = t^1.4)
    const u = Math.pow(t, 1.4);
    st.drive = F.drive ?? bmix(0.6, 1, u); st.hoard = F.hoard ?? bmix(60, 0, u); st.lateBoost = F.lateBoost ?? (1 - u) * 10;
    c.n2cap = F.cap ?? bmix(0.62, 0.82, u);         // a lower top speed (and bots never drive flat out: they'd over-run the ball)
    c.n2pow = F.pow ?? bmix(0.62, 1, u);             // softer touches at low ranks
    st.u = u;
    lockCar(c);
    if (st.act) { if (runAct(b, c, t, atk, dt, inp)) return inp; st.act = null; }
    st.think -= dt;
    // decisions every ~0.2 s for everyone; skill is how soon they act on one: the reaction time (Bronze 0.45 s ...
    // Legend 0.04 s), so a low-rank bot keeps playing the old read for a while after the ball changes
    const q = st.q || (st.q = []);
    if (st.think <= 0 || (!q.length && st.plan && st.plan.at < botClock - 0.05)) {
      st.think = (F.think || 0.2) * (0.75 + 0.5 * Math.random());
      q.push({ at: botClock + (F.delay ?? bmix(0.45, 0.04, st.u)), R: decide(b, c, t, atk) });
    }
    while (q.length && q[0].at <= botClock) Object.assign(st, q.shift().R);
    // in the air with nothing planned: land on the wheels
    if (!c.ground) {
      inp.n2.z = spinTo(c, upA(c), PI / 2);
      if (c.p.y > 6 && c.v.y < 0 && st.save && t > 0.5 && c.boost > 10) inp.boost = false;
      return inp;
    }
    // on a wall or the ceiling without a reason to be there: jump off
    if (c.n.y < 0.55 && !(Math.abs(ball.p.z - c.p.z) < 9 && ball.p.y > c.p.y)) { if (st.faceCD <= 0) { st.faceCD = 0.5; inp.jp = true; inp.jh = true; } return inp; }
    const bz = ball.p.z, by = ball.p.y, distZ = bz - c.p.z, spd = Math.abs(c.v.z);
    // kickoff: the first man goes, flips into the ball at the end; the others wait back
    if (st.kick) {
      if (st.role === 0) {
        // straight through it with boost (a flip at the ball loses the 50/50 in a single lane)
        // (a beat's pause off the line: in a single lane the second car to the ball tends to win the kickoff)
        if (st.kickWait == null) st.kickWait = Math.random() * (F.kickW ?? 0.36);
        if (st.kickWait > 0) { st.kickWait -= dt; return inp; }
        driveTo(b, c, 0, inp, { rush: true, keep: F.kickKeep ?? 0, boostD: 0 });
      } else driveTo(b, c, c.p.z, inp, { face: atk, keep: 100, boostD: 99 });
      return inp;
    }
    st.kickWait = null;
    const goalSide = (c.p.z - bz) * atk < 0.8;
    if (st.role === 0 || st.save) {
      const plan = st.plan;
      if (!goalSide) {
        // beaten: get back past the ball (under a high one; a low one gets played back), then play it again
        driveTo(b, c, bz - atk * 6, inp, { rush: true, keep: 10, boostD: 6 });
        return inp;
      }
      // nothing reachable: a save goes back to the goal line, otherwise follow the ball from our side
      if (!plan) { const tz = st.save ? ownZ + atk * 2 : bz - atk * (BR + 2); driveTo(b, c, tz, inp, { rush: st.save && Math.abs(tz - c.p.z) > 6, face: atk, keep: 30, boostD: 14 }); return inp; }
      const left = plan.at - botClock;
      const az = plan.z - atk * (BR + 1.1 * CAR_K) + st.noise;
      // low ranks hesitate: pull up short of the ball and let it come (or let the opponent have it)
      if (st.wrong) { driveTo(b, c, az - atk * 2.5, inp, { keep: 100, boostD: 99 }); return inp; }
      if (st.shadow) {
        const hold = ownZ + atk * 3.5;
        if (Math.abs(distZ) < 5 + spd * 0.15 && by < 4) driveTo(b, c, bz + atk * 2, inp, { rush: true, keep: 0, boostD: 0 });
        else driveTo(b, c, hold, inp, { face: atk, keep: 40, boostD: 8, rush: (c.p.z - hold) * atk > 5 });
        return inp;
      }
      // placed shots (higher ranks, more often): near their goal a rolling ball is pushed in at a controlled speed
      // instead of blasted (a hard hit there flies over the goal and comes back off the wall)
      if (plan.kind === 0 && by < 3 && A2.L - plan.z * atk < 12 && (c.p.z - bz) * atk < 0 && st.placeOn) {
        driveTo(b, c, az + atk * 3, inp, { keep: 100, boostD: 99, rush: true });
        inp.n2.z *= 0.65; inp.boost = false;
        return inp;
      }
      if (plan.kind === 0) {
        driveTo(b, c, az + atk * 3, inp, { rush: true, keep: F.atkKeep ?? (t > 0.6 ? 15 : 0), boostD: 10 });
        // close in, nobody contesting it: now and then a flip into it for a harder shot
        const gap = Math.abs(distZ) - BR - 1.3 * CAR_K, closing = c.v.z * sgn(distZ) - ball.v.z * sgn(distZ);

        return inp;
      }
      // in the air: be under it, then jump (or take off) at the right moment
      driveTo(b, c, az, inp, { rush: Math.abs(az - c.p.z) > Math.max(6, left * 12), keep: plan.kind === 2 ? 40 : 15, boostD: 12 });
      const err = (Math.random() - 0.5) * (1 - (F.errT ?? t)) * 0.25, hz = plan.z - c.p.z, toward = sgn(hz) || atk;
      // only off the ground, pointing and rolling the right way
      const ready = c.ground && c.n.y > 0.8 && (facing(c) === toward || Math.abs(hz) < 2) && c.v.z * toward > -3;
      if (!ready) return inp;
      if (plan.kind === 1) {
        const rise = botRise(plan.y - 1.2);
        if (rise > 0 && left <= rise + 0.03 + err && Math.abs(az - c.p.z) < 2 + spd * left && !F.noJump) startAct(st, "jump", {});
      } else {
        const need = (plan.y - 1) / 13 + 0.35;
        if (left <= need + err + 0.1 && Math.abs(hz) < (Math.max(0, c.v.z * toward) + 12) * left + 2 && !F.noAerial) startAct(st, "aerial", { at: plan.at, T: left, dbl: t > 0.75 && plan.y > 14 });
      }
      return inp;
    }
    // support: behind the play, between the ball and our goal, ready to follow up; keeper: in goal
    if (st.role === 1) {
      const back = F.back ?? bmix(9, 7, t), tz = atk > 0 ? Math.max(ownZ + 3, Math.min(bz - back, A2.L * 0.3)) : Math.min(ownZ - 3, Math.max(bz + back, -A2.L * 0.3));

    } else driveTo(b, c, ownZ + atk * 2.5, inp, { face: atk, keep: 60, boostD: 8 });
    return inp;

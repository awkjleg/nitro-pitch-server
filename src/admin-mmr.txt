// Admin: give or take away MMR, in the same panel as Give XP. The server (/api/progress admin-mmr, admin account
// only) changes the player's rating at once; their game shows "+50 MMR in Doubles from the admin" next time it loads.
(() => {
  const api = window.NPProgressAPI, PL = [[1, "Duel 1v1"], [2, "Doubles 2v2"], [3, "Standard 3v3"]];
  const xg = xpGiver;
  xpGiver = function (uid, name) {
    const box = xg(uid, name);
    const row = h("div", "xp-give mmr-give");
    const pl = document.createElement("select"); pl.className = "t"; for (const [n, t] of PL) pl.append(new Option(t, n));
    const amt = document.createElement("input"); amt.type = "number"; amt.min = "1"; amt.max = "4000"; amt.step = "1"; amt.placeholder = "MMR"; amt.className = "t";
    const note = document.createElement("input"); note.type = "text"; note.maxLength = 120; note.placeholder = "Message (optional)"; note.className = "t note";
    const give = async (v, sign) => {
      const n = Math.floor(Math.abs(+v)), p = PL.find(x => x[0] === +pl.value);
      if (!(n >= 1 && n <= 4000)) { adMsg("Pick an MMR amount from 1 to 4000."); return; }
      if (!confirm((sign > 0 ? "Give +" : "Remove ") + n + " MMR " + (sign > 0 ? "to " : "from ") + name + " in " + p[1] + "?")) return;
      try {
        const r = await api.call("admin-mmr", { id: crypto.randomUUID(), uid, pl: p[0], amount: n * sign, note: note.value.trim() });
        try { window.npPoke && window.npPoke(uid); } catch {}
        const t = name + " is now at " + r.mmr + " MMR in " + p[1]; toast(t); adMsg(t + "."); amt.value = ""; note.value = "";
      } catch (e) { adMsg((e && e.message) || "Couldn't change the MMR"); }
    };
    row.append(h("b", "", "MMR:"), pl, adBtn("+50", () => give(50, 1), true), adBtn("+100", () => give(100, 1), true), adBtn("−50", () => give(50, -1)), amt, adBtn("Give", () => give(amt.value, 1), true), adBtn("Remove", () => give(amt.value, -1)), note);
    const wrap = h("div", ""); wrap.append(box, row);
    Object.defineProperty(wrap, "hidden", { get: () => box.hidden, set: v => { box.hidden = v; row.hidden = v; } });
    return wrap;
  };
  // the player's side: show the admin's MMR change once
  const sp = sanitizeProfile;
  sanitizeProfile = function (s) { const o = sp(s); if (s && s.mmrGift && typeof s.mmrGift === "object") o.mmrGift = { amount: s.mmrGift.amount | 0, pl: s.mmrGift.pl | 0, note: String(s.mmrGift.note || "").slice(0, 120), at: +s.mmrGift.at || 0 }; return o; };
  setInterval(() => {
    const g = profile.mmrGift; if (!g || !g.at) return;
    let seen = 0; try { seen = +localStorage.getItem("np-mmr-seen") || 0; } catch {}
    if (g.at <= seen) return;
    try { localStorage.setItem("np-mmr-seen", String(g.at)); } catch {}
    const p = PL.find(x => x[0] === g.pl);
    toast((g.amount > 0 ? "+" + g.amount : String(g.amount)) + " MMR in " + (p ? p[1] : "ranked") + " from the admin" + (g.note ? ": " + g.note : ""));
  }, 3000);
  // Give items: a search box that finds any item by name across every type (cars, decals, wheels...); clicking a
  // result picks its type and item in the menus below
  const ra = renderAdmin;
  renderAdmin = async function () {
    const r = await ra.apply(this, arguments);
    try {
      const box = adTab === "give" && document.querySelector("#adminPop .ad-give");
      if (!box || box.querySelector(".np-isearch")) return r;
      const [, kindSel, itemSel] = box.querySelectorAll("select"); if (!kindSel || !itemSel) return r;
      const wrap = h("div", "np-isearch"), q = document.createElement("input"), res = h("div", "np-ires");
      q.type = "search"; q.className = "t"; q.placeholder = "Search items (e.g. titanium, octane, flag)"; q.autocomplete = "off"; q.setAttribute("aria-label", "Search items");
      const NOSTART = ["wheel", "topper", "antenna", "title", "banner"];
      const show = () => {
        res.innerHTML = ""; const t = q.value.trim().toLowerCase(); if (!t) return;
        const hits = [];
        for (const [k, kn] of GKINDS) listOf(k).forEach((it, i) => { if (NOSTART.includes(k) && i === 0) return; if (it && !it.none && String(it.name || "").toLowerCase().includes(t)) hits.push([k, kn, i, it]); });
        if (!hits.length) { res.append(h("small", "", "No items match \"" + q.value.trim() + "\".")); return; }
        for (const [k, kn, i, it] of hits.slice(0, 40)) {
          const b = document.createElement("button"); b.type = "button"; b.className = "np-ihit";
          b.append(h("b", "", it.name), h("small", "", kn + " · " + rarName(it) + (it.admin ? " · admin design" : "")));
          b.onclick = () => { kindSel.value = k; if (kindSel.onchange) kindSel.onchange(); itemSel.value = String(i); res.innerHTML = ""; q.value = it.name; adMsg("Picked " + it.name + " (" + kn + "). Choose the player and press Give item."); };
          res.append(b);
        }
        if (hits.length > 40) res.append(h("small", "", (hits.length - 40) + " more: type more of the name."));
      };
      q.addEventListener("input", show);
      wrap.append(q, res);
      const typeRow = kindSel.closest("label") || kindSel; box.insertBefore(wrap, typeRow);
    } catch {}
    return r;
  };
  { const st = document.createElement("style"); st.textContent = ".np-isearch{display:grid;gap:6px;margin:6px 0}.np-ires{display:grid;gap:4px;max-height:240px;overflow-y:auto}.np-ihit{display:flex;justify-content:space-between;gap:10px;align-items:baseline;text-align:left;padding:7px 10px;border-radius:8px;border:1px solid rgba(150,180,255,.25);background:rgba(0,0,0,.3);color:#e8eeff;font:inherit;cursor:pointer}.np-ihit:hover,.np-ihit:focus{background:rgba(90,140,255,.22)}.np-ihit small{color:#9fb0d4;font-size:12px}.np-ires>small{color:#9fb0d4}"; document.head.append(st); }
  // ---------- search boxes: players in Give items, the Players online list, and the admin's Players / Online now tabs ----------
  const npFind = (ph, onIn) => { const q = document.createElement("input"); q.type = "search"; q.className = "t np-find"; q.placeholder = ph; q.autocomplete = "off"; q.spellcheck = false; q.setAttribute("aria-label", ph); q.addEventListener("input", () => onIn(q.value.trim().toLowerCase())); return q; };
  const hasTxt = (el, t) => !t || String(el && el.textContent || "").toLowerCase().includes(t);
  const adQ = { players: "", online: "" };
  const ra2 = renderAdmin;
  renderAdmin = async function () {
    const r = await ra2.apply(this, arguments);
    try {
      const body = $("adBody"); if (!body || body.querySelector(".np-find")) return r;
      if (adTab === "give") {
        // the player menu: typing narrows it to matching names and picks the first one
        const who = body.querySelector(".ad-give select"); if (!who) return r;
        const q = npFind("Search players", t => {
          let first = null;
          for (const o of who.options) { if (!o.value) continue; const ok = hasTxt(o, t); o.hidden = !ok; if (ok && !first) first = o; }
          if (t && first) who.value = first.value; else if (t) who.value = "";
        });
        const lab = who.closest("label") || who; lab.parentNode.insertBefore(q, lab);
      } else if (adTab === "players") {
        const t = body.querySelector("table"); if (!t) return r;
        const apply = x => { adQ.players = x; for (const tr of t.rows) { if (!tr.cells.length || tr.cells[0].tagName === "TH" || tr.cells[0].colSpan === 7) continue; const ok = hasTxt(tr.cells[0], x), g = tr.nextElementSibling; tr.classList.toggle("np-hide", !ok); if (g && g.cells[0] && g.cells[0].colSpan === 7 && !ok) g.hidden = true; } };
        const q = npFind("Search players by name", apply); q.value = adQ.players; body.insertBefore(q, t); if (adQ.players) apply(adQ.players);
      } else if (adTab === "online") {
        const apply = x => { adQ.online = x; for (const d of body.querySelectorAll(":scope > .pm")) { const ok = hasTxt(d.querySelector(".who"), x), g = d.nextElementSibling; d.classList.toggle("np-hide", !ok); if (g && g.npUid && !ok) g.hidden = true; } };
        const q = npFind("Search online players", apply); q.value = adQ.online; body.prepend(q); if (adQ.online) apply(adQ.online);
      }
    } catch {}
    return r;
  };
  // the Players online list re-draws whenever someone comes or goes, so the search is applied again after each re-draw
  { const ol = $("onlineList");
    if (ol) {
      let olQ = "";
      const apply = () => { for (const row of ol.querySelectorAll(".pm.ol")) row.classList.toggle("np-hide", !hasTxt(row.querySelector(".who b") || row.querySelector(".who"), olQ)); };
      const q = npFind("Search players", x => { olQ = x; apply(); });
      q.style.cssText = "margin:8px 10px 4px;width:calc(100% - 20px);box-sizing:border-box";
      ol.parentNode.insertBefore(q, ol);
      new MutationObserver(() => { if (olQ) apply(); }).observe(ol, { childList: true });
    } }
  { const st = document.createElement("style"); st.textContent = ".np-hide{display:none!important}.np-find{display:block;width:100%;box-sizing:border-box;margin:4px 0 8px;padding:8px 10px;border-radius:8px;border:1px solid rgba(150,180,255,.35);background:rgba(0,0,0,.35);color:#fff;font:inherit}"; document.head.append(st); }
  // ---------- announcements: on everyone's screen the moment the admin publishes (the page listens to it), and bigger ----------
  { const st = document.createElement("style");
    st.textContent = "#annBar{font-size:clamp(17px,2.2vw,22px);line-height:1.35;padding:14px 12px 14px 22px;gap:16px;border-width:2px;border-radius:14px;width:max-content;max-width:min(920px,calc(100% - 32px));box-shadow:0 0 0 4px rgba(255,212,81,.14),0 16px 40px rgba(0,0,0,.55)}#annBar #annText{overflow-wrap:anywhere}";
    document.head.append(st);
    const show = x => {
      const text = x && typeof x.text === "string" ? x.text.trim() : "";
      let seen = ""; try { seen = localStorage.getItem("np-ann-closed") || ""; } catch {}
      if (text && seen !== text) { $("annText").textContent = text; $("annBar").hidden = false; } else if (!text || $("annText").textContent === text) $("annBar").hidden = true;
    };
    const A = window.npAccounts;
    Promise.resolve(A && A.ready).catch(() => {}).then(() => {
      try { firebase.firestore().collection("config").doc("announce").onSnapshot(d => show(d && d.exists ? d.data() : null), () => {}); } catch {}
    });
  }
  // ---------- admin: see a player's items and click one to take it away (in the same panel as Give XP) ----------
  const xg2 = xpGiver;
  xpGiver = function (uid, name) {
    const box = xg2(uid, name), row = h("div", "xp-give inv-take"), grid = h("div", "np-inv");
    grid.hidden = true;
    const load = async () => {
      grid.hidden = false; grid.innerHTML = ""; grid.append(h("small", "", "Loading " + name + "'s items…"));
      let inv, adminList = [];
      try { inv = await api.call("admin-inventory", { id: crypto.randomUUID(), uid }); } catch (e) { grid.innerHTML = ""; grid.append(h("small", "", (e && e.message) || "Couldn't load their items")); return; }
      try { const A = window.npAccounts; if (A && A.adminItems) adminList = await A.adminItems(uid); } catch {}
      grid.innerHTML = "";
      const items = [];
      for (const [kind] of GKINDS) for (const i of (inv.own && inv.own[kind]) || []) if (listOf(kind)[i]) items.push([kind, i, false]);
      for (const key of adminList) { const m = /^([a-z]+):(\d+)$/.exec(key); if (m && listOf(m[1]) && listOf(m[1])[+m[2]]) items.push([m[1], +m[2], true]); }
      const head = h("small", "", items.length ? name + " has " + items.length + " item" + (items.length === 1 ? "" : "s") + " (and " + ((inv.drops || []).length) + " unopened drops). Click one to take it away." : name + " has no items from drops, gifts or trades.");
      grid.append(head);
      for (const [kind, i, adm] of items) {
        const it = listOf(kind)[i];
        const t = tradeTile(kind, i, async () => {
          if (!confirm("Take " + it.name + " (" + kindName(kind) + ") away from " + name + "?")) return;
          try {
            if (adm) await firebase.firestore().collection("adminItems").doc(String(uid)).set({ items: firebase.firestore.FieldValue.arrayRemove(kind + ":" + i) }, { merge: true });
            else await api.call("admin-take", { id: crypto.randomUUID(), uid, kind, i });
            t.remove(); try { window.npPoke && window.npPoke(uid); } catch {}
            toast("Took " + it.name + " from " + name); adMsg("Took " + it.name + " from " + name + ".");
          } catch (e) { toast((e && e.message) || "Couldn't take it away"); }
        });
        if (adm) t.title = "ADMIN design item"; grid.append(t);
      }
      try { (function pump() { if (grid.isConnected && !grid.hidden) { try { if (thumbs.pending) thumbs.pump(); } catch {} requestAnimationFrame(pump); } })(); } catch {}
    };
    row.append(h("b", "", "Items:"), adBtn("Show items", () => { if (!grid.hidden) { grid.hidden = true; return; } load(); }), adBtn("Refresh", load));
    const wrap = h("div", ""); wrap.append(box, row, grid);
    Object.defineProperty(wrap, "hidden", { get: () => box.hidden, set: v => { box.hidden = v; row.hidden = v; if (v) grid.hidden = true; } });
    return wrap;
  };
  { const st = document.createElement("style"); st.textContent = ".np-inv{display:grid;grid-template-columns:repeat(auto-fill,minmax(120px,1fr));gap:6px;margin:6px 0 10px;max-height:340px;overflow-y:auto}.np-inv>small{grid-column:1/-1;color:#9fb0d4}.np-inv .tr-it:hover{outline:2px solid #e74c3c}"; document.head.append(st); }
})();

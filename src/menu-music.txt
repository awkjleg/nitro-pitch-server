// Menu music: real songs (public/music/*.mp3) instead of the built-in synth loop. The game's sound engine calls
// npMenuTracks from its music tick with its own music gain, so the Settings music volume, the mute button and the
// fade-out during matches all still apply. Each song is downloaded whole and decoded into the sound engine (no <audio>
// element: those need the server to answer range requests, and Safari won't play them otherwise). If a song can't
// load, the old synth loop plays instead.
const NP_TRACKS = ["/music/nitro-pitch-1.mp3", "/music/nitro-pitch-2.mp3"];
let npTrackFailed = false, npTrackN = Math.floor(Math.random() * NP_TRACKS.length), npTrackSrc = null, npTrackBusy = false, npTrackStop = 0;
const npTrackBufs = {};
function npTrackLoad(ctx, i) {
  return npTrackBufs[i] || (npTrackBufs[i] = fetch(NP_TRACKS[i]).then(r => { if (!r.ok) throw new Error("music " + r.status); return r.arrayBuffer(); })
    .then(b => new Promise((res, rej) => ctx.decodeAudioData(b, res, rej))));
}
function npMenuTracks(ctx, dest, playing) {
  if (npTrackFailed) return false;
  // music off, muted or a match on: stop the song once the engine's fade-out is done
  if (!playing) { if (npTrackSrc && !npTrackStop) npTrackStop = setTimeout(() => { const s = npTrackSrc; npTrackSrc = null; npTrackStop = 0; try { s.stop(); } catch {} }, 1500); return true; }
  clearTimeout(npTrackStop); npTrackStop = 0;
  if (npTrackSrc || npTrackBusy) return true;
  npTrackBusy = true;
  npTrackLoad(ctx, npTrackN).then(buf => {
    npTrackBusy = false;
    if (npTrackSrc) return;
    const s = ctx.createBufferSource(), g = ctx.createGain(); g.gain.value = 0.5; s.buffer = buf; s.connect(g).connect(dest);
    s.onended = () => { if (npTrackSrc === s) { npTrackSrc = null; npTrackN = (npTrackN + 1) % NP_TRACKS.length; } };
    s.start(); npTrackSrc = s;
    npTrackLoad(ctx, (npTrackN + 1) % NP_TRACKS.length).catch(() => {});
  }).catch(() => { npTrackBusy = false; npTrackFailed = true; });
  return true;
}

// ---- goal speed and goal time: "87 MPH · 1:12 into the game" under every GOAL! (worked out on each screen when the goal
// is shown: ball speed in uu/s is 50x the game's units, 1 uu/s = 0.0223694 mph; the time is the match length minus the clock)
let npGoalKey = "", npGoalText = "";
function npGoalLine(sub) {
  const key = match.mid + ":" + match.s[0] + ":" + match.s[1];
  if (key !== npGoalKey) {
    npGoalKey = key;
    const mph = Math.round(ball.v.length() * 50 * 0.0223694), total = LENGTHS[match.mut[0] | 0];
    let when = "";
    if (match.ot) when = "in overtime";
    else if (Number.isFinite(total) && !freePlay && fin(match.tl)) { const t = Math.max(0, Math.round(total - match.tl)); when = Math.floor(t / 60) + ":" + String(t % 60).padStart(2, "0") + " into the game"; }
    npGoalText = mph + " MPH" + (when ? " · " + when : "");
  }
  return (sub ? sub + " · " : "") + npGoalText;
}
